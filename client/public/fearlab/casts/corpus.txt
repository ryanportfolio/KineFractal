# ---------------------------------------------------------------- aux-cache guard
#
# Exo trim signals read GITIGNORED daily caches (fearlab/data/*.csv) that a fresh
# worktree/clone does NOT have. Each loader (_load_credit/_load_participation/
# _load_dxy/_load_soxlead/_load_djtlead/_load_tipief/_load_iposlead) falls back to
# {} on a missing file -> the signal runs SILENTLY INERT and the backtest emits
# plausible-but-WRONG numbers with no error. This map + ensure_aux() close that
# hole: simulate() refuses to run an enabled signal whose cache is absent
# (Layer 1), fetching it first when possible (Layer 2).
#
# MAINTENANCE: every exo loader MUST register its daily deps here, INCLUDING the
# cross-symbol ratio denominator (e.g. SOX/QQQ needs QQQ even on a SPY backtest).
# Symbols are in data.get_daily form (^ and / become _ on disk). sync_caches.py
# --check asserts this covers refresh.AUX_FEEDS so a new signal cannot silently
# reopen the hole.
AUX_REQUIRED = {
    "credSellOn": ("HYG", "IEF"),
    "pbSellOn":   ("RSP", "SPY"),
    "dxySellOn":  ("DX-Y.NYB",),
    "soxSellOn":  ("^SOX", "QQQ"),
    "djtSellOn":  ("^DJT", "SPY"),
    "tipSellOn":  ("TIP", "IEF"),
    "ipoSellOn":  ("IPO", "SPY"),
}

_AUX_ENSURED: set = set()   # per-process memo: fetch each missing feed at most once


def _required_aux(p: dict) -> set:
    need: set = set()
    for flag, syms in AUX_REQUIRED.items():
        if p.get(flag):
            need.update(syms)
    return need


def ensure_aux(p: dict, fetch: bool = True) -> None:
    """Guard against silently-inert exo signals caused by missing daily caches.

    For every aux/cross-symbol daily an ENABLED signal needs: if its cache is
    absent, fetch it (when fetch=True, via data.get_daily's curl path) and, if it
    still cannot be obtained, RAISE rather than let the signal run inert. Cheap
    no-op when caches are present (one stat per feed). Called at the top of
    simulate(); also runnable standalone via sync_caches.py."""
    import data
    missing = []
    for sym in sorted(_required_aux(p)):
        path = data._cache_path(sym, "1d")
        if os.path.exists(path):
            continue
        if fetch and sym not in _AUX_ENSURED:
            _AUX_ENSURED.add(sym)
            try:
                data.get_daily(sym)          # auto-fetches a missing/stale cache
            except Exception:
                pass
        if not os.path.exists(path):
            missing.append((sym, os.path.basename(path)))
    if missing:
        lst = "; ".join(f"{s} -> {fn}" for s, fn in missing)
        raise SystemExit(
            "FearLab aux-cache guard: missing required cache(s) for enabled "
            f"signals: {lst}.\n"
            "  These are gitignored (absent in a fresh worktree) -> the signal "
            "would run SILENTLY INERT and produce WRONG numbers.\n"
            "  Fix: `py fearlab/sync_caches.py` (online) or copy fearlab/data/"
            "*.csv from the main checkout.")

def collect_fired_engines(pending_buys: list[dict], pending_sells: list[dict],
                          working_limits: dict) -> list[str]:
    """Sorted, deduped set of order "tag"s still pending at the END of simulate's
    loop. At loop end these three broker structures hold ONLY the orders queued on
    the decision bar (li) that would fill at the next open -> their tags ARE the
    engines that fired on the decision bar. Pure / read-only: no broker state is
    mutated and nothing inside the simulate loop is touched, so the backtest board
    is byte-identical. The execution bridge uses this to detect any engine that
    fires in-sim but has no intent handler / deferred entry, then loudly defer/HALT
    instead of silently dropping the order. NB sell tags ("Protect","Trim z1.1",
    "IpoTrim",...) use a DIFFERENT vocabulary from intent.py engine names
    ("Protect","z_tier1",...); the bridge normalizes before matching. See
    fearlab/bridge/intent.py."""
    fired: set[str] = set()
    for o in pending_buys:
        if o.get("tag"):
            fired.add(o["tag"])
    for o in pending_sells:
        if o.get("tag"):
            fired.add(o["tag"])
    for w in working_limits.values():
        if w.get("tag"):
            fired.add(w["tag"])
    return sorted(fired)


def collect_grid_buys(pending_buys: list[dict], params: dict) -> list[dict]:
    """The passive accumulation grid (tiers 1..5) queued on the decision bar.

    The sim queues each enabled+triggered tier as
    br.queue_buy(f"Fear T{n}", "grid", qty=equity*params[f"b{n}Qty"]/100/c, ...).
    At simulate's loop end br.pending_buys holds ONLY the orders queued on the
    decision bar (li) that fill at next open (= MOO on a 1d cell). So a tier is
    present here IFF the sim queued it on li -> every sim gate (b{n}En, buy_sig
    crossover, can_buy_passive, broken-protect lock) is self-respected by
    construction; no predicate is re-derived. qty_pct = params[f"b{n}Qty"] is the
    exact scale-invariant size the sim used. A tier queues at most once per bar,
    but we collapse to one entry per tier defensively. Pure / read-only.

    Returns [{"tier": n, "qty_pct": float}, ...] sorted by tier."""
    tiers: dict[int, float] = {}
    for o in pending_buys:
        tag = o.get("tag") or ""
        if tag.startswith("Fear T"):
            try:
                n = int(tag[len("Fear T"):])
            except ValueError:
                continue
            if 1 <= n <= 5:
                qty = params.get(f"b{n}Qty")   # defensive: skip a tier whose Qty key is absent
                if qty is not None:
                    tiers[n] = float(qty)
    return [{"tier": n, "qty_pct": tiers[n]} for n in sorted(tiers)]

# ---------------------------------------------------------------- HTF helpers

def _date(ts: int) -> datetime.date:
    t = time.gmtime(ts)
    return datetime.date(t.tm_year, t.tm_mon, t.tm_mday)


def _group_daily(daily: Bars, key_fn) -> tuple[Bars, list[int]]:
    """Aggregate daily bars into HTF bars by key. Returns (htf_bars, daily->htf idx)."""
    htf = Bars()
    day_to_htf: list[int] = []
    cur_key = None
    for i in range(len(daily)):
        k = key_fn(_date(daily.ts[i]))
        if k != cur_key:
            cur_key = k
            htf.append(daily.ts[i], daily.open[i], daily.high[i], daily.low[i], daily.close[i], daily.volume[i])
        else:
            j = len(htf) - 1
            htf.high[j] = max(htf.high[j], daily.high[i])
            htf.low[j] = min(htf.low[j], daily.low[i])
            htf.close[j] = daily.close[i]
            htf.volume[j] += daily.volume[i]
        day_to_htf.append(len(htf) - 1)
    return htf, day_to_htf


def _map_chart_to_htf(chart: Bars, htf: Bars) -> list[int]:
    """Forming HTF index per chart bar (rightmost htf bar with ts <= chart ts)."""
    out = []
    k = 0
    n = len(htf)
    for i in range(len(chart)):
        while k + 1 < n and htf.ts[k + 1] <= chart.ts[i]:
            k += 1
        out.append(k)
    return out


def _build_htf_raw(tf: str, chart: Bars, daily: Bars) -> tuple[Bars, list[int]]:
    """Returns (htf bars, chart->forming-htf-idx). tf: '240'|'D'|'W'|'M'|'3M'."""
    if tf == "240":
        htf = resample(chart, 14400)
        return htf, _map_chart_to_htf(chart, htf)
    if tf in ("D", "1D"):
        htf = daily
    elif tf == "W":
        htf, _ = _group_daily(daily, lambda d: d.isocalendar()[:2])
    elif tf in ("M", "1M"):
        htf, _ = _group_daily(daily, lambda d: (d.year, d.month))
    elif tf == "3M":
        htf, _ = _group_daily(daily, lambda d: (d.year, (d.month - 1) // 3))
    else:
        raise SystemExit(f"unsupported HTF: {tf}")
    return htf, _map_chart_to_htf(chart, htf)


def _stdev(src: list[float], length: int) -> list[float]:
    """Pine ta.stdev (population)."""
    out = [NAN] * len(src)
    s = s2 = 0.0
    q: list[float] = []
    for i, x in enumerate(src):
        if is_na(x):
            continue
        q.append(x)
        s += x
        s2 += x * x
        if len(q) > length:
            old = q.pop(0)
            s -= old
            s2 -= old * old
        if len(q) == length:
            m = s / length
            var = max(s2 / length - m * m, 0.0)
            out[i] = math.sqrt(var)
    return out


def _crossover(cur: float, prev: float, th: float) -> bool:
    return (not is_na(cur)) and (not is_na(prev)) and prev <= th < cur


def _crossunder(cur: float, prev: float, th: float) -> bool:
    return (not is_na(cur)) and (not is_na(prev)) and prev >= th > cur


def _is_near(price: float, level: float, atr: float, mod: float) -> bool:
    if is_na(level) or is_na(atr):
        return False
    return abs(price - level) <= atr * mod


def _nw_prepass_raw(src: list[float], bw: float, mult: float) -> tuple[list[float], list[float]]:
    """Endpoint Nadaraya-Watson + MAE bands (non-repainting). Same math as
    lab/strategy.py and the user's NWE alert indicator. Returns (out, mae)."""
    n = len(src)
    coefs = [math.exp(-(i * i) / (bw * bw * 2)) for i in range(500)]
    den = sum(coefs)
    out = [NAN] * n
    mae = [NAN] * n
    win: list[float] = []
    s = 0.0
    for i in range(n):
        cur = src[i]
        if i >= 499:
            w = src[i - 499:i + 1]
            o = sum(coefs[k] * w[499 - k] for k in range(500)) / den
        else:
            o = sum(coefs[k] * (src[i - k] if i - k >= 0 else cur) for k in range(500)) / den
        out[i] = o
        d = abs(cur - o)
        win.append(d)
        s += d
        if len(win) > 499:
            s -= win.pop(0)
        if len(win) == 499:
            mae[i] = s / 499 * mult
    return out, mae


def _wvf(close: list[float], low: list[float], pd_: int) -> list[float]:
    hi = ta.rolling_max(close, pd_)
    out = [NAN] * len(close)
    for i in range(len(close)):
        if is_na(hi[i]) or hi[i] == 0:
            continue
        out[i] = (hi[i] - low[i]) / hi[i] * 100
    return out


_BREADTH_CACHE: dict | None = None


def _load_breadth() -> dict:
    """NYSE breadth from TV exports (fearlab/data_ext/): per calendar date,
    (mcclellan, nh_pct, nl_pct, ratio_ok). Empty dict if files absent."""
    global _BREADTH_CACHE
    if _BREADTH_CACHE is not None:
        return _BREADTH_CACHE
    import csv
    ext = os.path.join(_HERE, "data_ext")

    def load(name):
        out = {}
        path = os.path.join(ext, name)
        if not os.path.exists(path):
            return out
        with open(path, newline="") as f:
            for row in csv.DictReader(f):
                try:
                    out[int(row["time"])] = float(row["close"])
                except (ValueError, KeyError):
                    pass
        return out

    adv, dec, unch = load("advn.csv"), load("decl.csv"), load("unch.csv")
    nh, nl = load("hign.csv"), load("lown.csv")
    common = sorted(set(adv) & set(dec) & set(unch) & set(nh) & set(nl))
    ratio = [(adv[t] - dec[t]) / (adv[t] + dec[t]) if adv[t] + dec[t] else 0.0
             for t in common]
    e19, e39 = ta.ema(ratio, 19), ta.ema(ratio, 39)
    out = {}
    for i, t in enumerate(common):
        if is_na(e19[i]) or is_na(e39[i]):
            continue
        total = adv[t] + dec[t] + unch[t]
        if total <= 100:
            continue
        out[_date(t)] = ((e19[i] - e39[i]) * 1000,
                         nh[t] / total * 100, nl[t] / total * 100,
                         nh[t] < nl[t] * 2)
    _BREADTH_CACHE = out
    return out

# decoupled from `window` (default 500): between RANK_MIN_OBS and `window` obs

def _div_prepass_raw(close: list[float], high: list[float], low: list[float],
                     volume: list[float], rsi_len: int, window: int,
                     wide: bool) -> list[int]:
    """Bullish-divergence activity (ported from the user's TV stack).
    Base families (Divergence Accumulator), causal with the Pine lags:
    RSI hidden/regular bull on the wick-sensitive curve (rsi of lows,
    fractal pivot confirmed 2 bars after the low) and PPO bull divergence
    (12/26/sma2, local minima confirmed 1 bar after, oscillator min within
    5 bars of the price min). `wide` (Divergence for Many / PPO Divergence
    long-term) adds MACD-hist, MFI, OBV and CCI(20) bull divergences via
    the same local-min pairing, plus a long-term PPO divergence: price
    prints a 55-bar low while the PPO bottom holds above its 55-bar low.
    Returns the per-bar count (0-3 base, 0-8 wide) of families detected
    within the trailing `window` bars."""
    n = len(close)
    rsi_b = ta.rsi(low, rsi_len)
    d = ta.ppo_smoothed(close, 12, 26, 2)
    oscs: dict[str, list[float]] = {"ppo": d}
    if wide:
        hlc3 = [(high[i] + low[i] + close[i]) / 3 for i in range(n)]
        oscs["macdh"] = ta.macd_hist(close, 12, 26, 9)
        oscs["mfi"] = ta.mfi(hlc3, volume, 14)
        oscs["obv"] = ta.obv(close, volume)
        tpm = ta.sma(hlc3, 20)
        cci = [NAN] * n
        for i in range(19, n):
            if is_na(tpm[i]):
                continue
            md = sum(abs(hlc3[i - k] - tpm[i]) for k in range(20)) / 20
            cci[i] = (hlc3[i] - tpm[i]) / (0.015 * md) if md > 0 else NAN
        oscs["cci"] = cci
    far = -(10 ** 9)
    last = {k: far for k in oscs}
    last["hid"] = far
    last["reg"] = far
    if wide:
        last["ppolt"] = far
    low55 = ta.rolling_min(low, 55) if wide else None
    d55 = ta.rolling_min(d, 55) if wide else None
    prev_rsi = prev_prc = NAN
    pm_bar = far
    cur_pl = prev_pl = NAN          # price lows at successive local minima
    cur_ol = {k: NAN for k in oscs}      # oscillator values at their minima
    prev_ol = {k: NAN for k in oscs}
    out = [0] * n
    for i in range(n):
        # RSI fractal bottom centered at i-2, confirmed at i
        if i >= 4 and not any(is_na(rsi_b[j]) for j in range(i - 4, i + 1)):
            if rsi_b[i - 4] > rsi_b[i - 2] and rsi_b[i - 3] > rsi_b[i - 2] \
                    and rsi_b[i - 2] < rsi_b[i - 1] and rsi_b[i - 2] < rsi_b[i]:
                cr, cp = rsi_b[i - 2], low[i - 2]
                if not is_na(prev_rsi):
                    if cp > prev_prc and cr < prev_rsi:
                        last["hid"] = i              # higher low / weaker RSI
                    elif cp < prev_prc and cr > prev_rsi:
                        last["reg"] = i              # lower low / stronger RSI
                prev_rsi, prev_prc = cr, cp
        if i >= 2:
            # price local min at i-1, confirmed at i
            if low[i] > low[i - 1] and low[i - 1] < low[i - 2]:
                cur_pl, prev_pl = low[i - 1], cur_pl
                pm_bar = i
                # long-term PPO div: 55-bar price low, PPO bottom held above
                if wide and not is_na(low55[i - 1]) and not is_na(d55[i]) \
                        and low[i - 1] <= low55[i - 1] \
                        and not is_na(cur_ol["ppo"]) and cur_ol["ppo"] > d55[i]:
                    last["ppolt"] = i
            # oscillator local min at i-1, confirmed at i (each family)
            for k, osc in oscs.items():
                if not any(is_na(osc[j]) for j in (i - 2, i - 1, i)) \
                        and osc[i] > osc[i - 1] and osc[i - 1] < osc[i - 2]:
                    cur_ol[k], prev_ol[k] = osc[i - 1], cur_ol[k]
                    if i - pm_bar <= 5 and not is_na(prev_pl) and not is_na(prev_ol[k]) \
                            and cur_pl < prev_pl and cur_ol[k] > prev_ol[k]:
                        last[k] = i
        out[i] = sum(1 for b in last.values() if i - b < window)
    return out

# _build_htf / _nw_prepass / _div_prepass are PURE functions of their args, but
# across a candidate sweep in one process they are recomputed identically on
# every simulate() call (and _build_htf 5-7x WITHIN a call). These wrappers cache
# by input identity so a warm process pays each once — behavior-preserving (the
# *_raw bodies are untouched). Keys use id() of the bar arrays PLUS a strong ref
# to the keyed object stored in the value (the `is` guard); holding the ref means
# a freed-then-reallocated object can never alias a stale entry (id reuse). NOTE:
# id() alone is unsafe because one process may evaluate multiple symbols (edge.py
# caches charts per (sym,tf)) — the bar-array identity is what distinguishes them.
# Process-lifetime caches, same convention as the _load_* caches above; bounded by
# the distinct (symbol, tf, prep-param) combinations a process touches.
_HTF_CACHE: dict = {}
_NW_CACHE: dict = {}
_DIV_CACHE: dict = {}


def _build_htf(tf: str, chart: Bars, daily: Bars) -> tuple[Bars, list[int]]:
    k = (tf, id(chart), id(daily))
    hit = _HTF_CACHE.get(k)
    if hit is not None and hit[0] is chart and hit[1] is daily:
        return hit[2]
    res = _build_htf_raw(tf, chart, daily)
    _HTF_CACHE[k] = (chart, daily, res)
    return res


def _nw_prepass(src: list[float], bw: float, mult: float) -> tuple[list[float], list[float]]:
    k = (id(src), bw, mult)
    hit = _NW_CACHE.get(k)
    if hit is not None and hit[0] is src:
        return hit[1]
    res = _nw_prepass_raw(src, bw, mult)
    _NW_CACHE[k] = (src, res)
    return res


def _div_prepass(close: list[float], high: list[float], low: list[float],
                 volume: list[float], rsi_len: int, window: int,
                 wide: bool) -> list[int]:
    k = (id(close), rsi_len, window, wide)
    hit = _DIV_CACHE.get(k)
    if hit is not None and hit[0] is close:
        return hit[1]
    res = _div_prepass_raw(close, high, low, volume, rsi_len, window, wide)
    _DIV_CACHE[k] = (close, res)
    return res

def _eff_trail(p: dict, atr: float, z: float, er: float,
               in_exo: bool) -> tuple[float, float, float]:
    """Effective per-bar lotTrail (arm, retrace, sell-frac) AFTER the ATR /
    z-slope / chop / exo-window modifiers — as fractions (0..1).

    SHARED by the sim loop (sec E2) and the live snapshot export (live['lot_trail'])
    so the two replays of the trail math can never drift. Pure; this is an exact
    extraction of the inline block in the E2 harvest loop — changing it changes
    BOTH callers in lockstep. `in_exo` = the exo-window TIME predicate, evaluated by
    the caller (loop: current day_count; export: final day_count); the
    lotTrailExoWindowDays>0 feature-gate is applied INSIDE here too, so the exo
    multiplier can never fire on a cell where the feature is off."""
    arm = float(p["lotTrailArmPct"]) / 100
    retr = float(p["lotTrailRetracePct"]) / 100
    if p["lotTrailAtrOn"] and not is_na(atr):
        retr = float(p["lotTrailAtrK"]) * atr
    tfrac = float(p["lotTrailSellFrac"]) / 100
    zslope = float(p["lotTrailFracZSlope"])
    if zslope > 0 and not is_na(z) and z > 0:
        tfrac = min(1.0, (float(p["lotTrailSellFrac"]) + zslope * z) / 100)
    if float(p["lotTrailChopErMax"]) > 0 and not is_na(er) \
            and er < float(p["lotTrailChopErMax"]):
        arm = float(p["lotTrailChopArmPct"]) / 100
        retr = float(p["lotTrailChopRetracePct"]) / 100
        tfrac = float(p["lotTrailChopSellFrac"]) / 100
    if in_exo and int(p["lotTrailExoWindowDays"]) > 0:
        retr *= float(p["lotTrailExoRetraceX"])
    return arm, retr, tfrac


def simulate(chart: Bars, daily: Bars, params: dict | None = None,
             xfear: dict | None = None, dividends: dict | None = None) -> dict:
    """xfear: optional {daily_ts: wvf_percentile} for ANOTHER symbol —
    enables the favXFearBoost cross-symbol fear sizing boost.
    dividends: optional {ex_date_epoch: per_share_amount} — with divOn,
    credits dividends to BOTH sides (strategy: cash, joins the pool and
    deploys per the normal rules; benchmark: DRIP at the ex-date open).
    Measurement-honesty diagnostic: the raw-close sim otherwise favors
    the strategy, whose average exposure is below the bench's 100%."""
    p = dict(DEFAULTS)
    if params:
        for k, v in params.items():
            if k not in DEFAULTS:
                raise SystemExit(f"unknown param: {k}")
            p[k] = v
    ensure_aux(p)                    # Layer 1+2: no silently-inert exo signals
    n = len(chart)
    if n < 50:
        raise SystemExit(f"not enough chart bars: {n}")

    # ---- backtest window (warmup bars before w0 trade nothing) ----
    w0, w1 = 0, n - 1
    sd, ed = str(p["startDate"]), str(p["endDate"])
    if sd:
        while w0 < n - 1 and _date(chart.ts[w0]).isoformat() < sd:
            w0 += 1
    if ed:
        while w1 > 0 and _date(chart.ts[w1]).isoformat() > ed:
            w1 -= 1
    if w1 - w0 < 50:
        raise SystemExit(f"window too small: {w1 - w0 + 1} bars ({sd!r}..{ed!r})")

    # ---- chart-TF indicator arrays ----
    atr_val = ta.atr(chart.high, chart.low, chart.close, int(p["atrLen"]))
    vol_sma = ta.sma_simple(chart.volume, int(p["volLen"]))
    zz_hi = ta.rolling_max(chart.high, int(p["zigzagLen"]))
    zz_lo = ta.rolling_min(chart.low, int(p["zigzagLen"]))
    sw_ph, sw_pl = ta.pivots(chart.high, 2, 2)
    sw_pl2 = ta.pivots(chart.low, 2, 2)[1]
    sw_h = [not is_na(x) for x in sw_ph]
    sw_l = [not is_na(x) for x in sw_pl2]

    bull_e = [False] * n
    bear_e = [False] * n
    for i in range(1, n):
        bear_e[i] = chart.close[i] < chart.open[i] and chart.open[i] > chart.high[i - 1] and chart.close[i] < chart.low[i - 1]
        bull_e[i] = chart.close[i] > chart.open[i] and chart.open[i] < chart.low[i - 1] and chart.close[i] > chart.high[i - 1]

    # ---- WVF (chart TF or daily-confirmed) ----
    wvf_tf = str(p["wvfTf"])
    if wvf_tf == "":
        wvf = _wvf(chart.close, chart.low, int(p["pd"]))
        wvf_prev = [NAN] + wvf[:-1]
        wvf_edge = [True] * n          # crossovers evaluated every chart bar
    else:
        d_wvf = _wvf(daily.close, daily.low, int(p["pd"]))
        _, c2d = _build_htf("D", chart, daily)
        wvf = [NAN] * n
        wvf_prev = [NAN] * n
        wvf_edge = [False] * n
        prev_k = -1
        for i in range(n):
            k = c2d[i]
            done = k - 1                 # completed daily bar
            if done >= 0:
                wvf[i] = d_wvf[done]
            if done >= 1:
                wvf_prev[i] = d_wvf[done - 1]
            wvf_edge[i] = k != prev_k    # first chart bar of a new day
            prev_k = k

    zlb = int(p["zLookback"])
    wvf_mean = ta.sma_simple(wvf, zlb)
    wvf_std = _stdev(wvf, zlb)
    wvf_z = [NAN] * n
    for i in range(n):
        if is_na(wvf[i]) or is_na(wvf_mean[i]) or is_na(wvf_std[i]):
            continue
        sd = wvf_std[i] if wvf_std[i] != 0 else 1.0
        wvf_z[i] = (wvf[i] - wvf_mean[i]) / sd
    wvf_pct = ta.percentrank(wvf, zlb)

    # ---- trend filter ----
    trend_sel = str(p["trendSel"])
    trend_line = [NAN] * n
    if trend_sel != "None":
        tf = "M" if trend_sel == "Monthly 5 SMA" else "3M"
        length = 5 if trend_sel == "Monthly 5 SMA" else 20
        htf, c2h = _build_htf(tf, chart, daily)
        htf_sma = ta.sma_simple(htf.close, length)
        for i in range(n):
            done = c2h[i] - 1            # sma(close,len)[1] inside security
            if done >= 0 and not is_na(htf_sma[done]):

        ts, o, h, l, c = chart.ts[i], chart.open[i], chart.high[i], chart.low[i], chart.close[i]
        in_win = w0 <= i < w1            # orders queued here fill by w1

        # bear lock: while the protect regime is broken (prot_broken here = prior-bar
        # latch state; section G updates it later this bar -> no lookahead), optionally
        # drop EVERY new buy except Reclaim -> go to cash and stay until the regime
        # reclaims. Default-inert (protectBrokenLockAll False keeps buy_block off).
        br.buy_block = bool(p.get("protectBrokenLockAll") and prot_broken)

        # -- month roll: external contribution lands before fills --
        if w0 <= i:
            mo = _date(ts).strftime("%Y-%m")
            if cur_month is None:
                cur_month = mo
            elif mo != cur_month:
                cur_month = mo
                if p["contribOn"] and p["contribUsd"] > 0:
                    br.contribute(ts, float(p["contribUsd"]))
                    bench_pending += float(p["contribUsd"])
                    if glide_n > 0:
                        glide_parcels.append([month_count + 1 + glide_n, float(p["contribUsd"])])
                month_count += 1
                if glide_n > 0:          # expired parcels force-invest now
                    expired = sum(parc[1] for parc in glide_parcels if parc[0] <= month_count)
                    glide_parcels = [parc for parc in glide_parcels if parc[0] > month_count]
                    if expired > 1.0:
                        br.queue_buy("Glide", "grid", expired / chart.close[i], "Glide",
                                     unlock_reserve=True)
                        count(signal_counts, "Glide")

        if i == w0:
            bench_pending += float(p["initialCapital"])
            signal_counts.clear()        # drop phantom warmup-bar counts
            sell_counts.clear()
            if glide_n > 0:              # lump amortizes over months 1..N
                per = float(p["initialCapital"]) / glide_n
                glide_parcels = [[m, per] for m in range(1, glide_n + 1)]

        # -- dividend credit (divOn): first bar of the ex-date, at open --
        if div_by_day and w0 <= i:
            dd = _date(ts)
            if dd != cur_div_day:
                cur_div_day = dd
                amt = div_by_day.get(dd)
                if amt:
                    shares = sum(lot.qty for lot in br.lots)
                    if shares > 0:
                        br.cash += shares * amt          # strategy: cash into the pool
                    if bench_qty > 0:
                        bench_qty += bench_qty * amt / o  # bench: DRIP at the open

        # -- fills --
        br.on_bar_open(i, ts, o)
        br.on_bar_intrabar(i, ts, o, h, l)
        if bench_pending > 0:
            bench_qty += bench_pending / o
            bench_pending = 0.0

        # -- fav limit-bid fallback (IF1-6): untouched after one bar -> market --
        if fav_limit is not None and i > fav_limit["bar"]:
            if "Fav" in br.working_limits:
                br.cancel_limit("Fav")
                br.queue_buy("Fav", "grid", fav_limit["qty"], "Fav",
                             unlock_reserve=fav_limit["unlock"])
                count(signal_counts, "FavLimitFallback")
            fav_limit = None

        # -- signal computation (bar close) --
        # trend
        is_uptrend = True if is_na(trend_line[i]) else c > trend_line[i]
        can_buy_passive = (trend_sel == "None") or is_uptrend

        # level latches (security value: pivot confirmation on completed HTF bar)
        for lv in ("l1", "l2", "l3"):
            if levels[lv] is None:
                continue
            done = levels[lv]["c2h"][i] - 1
            if done >= 0:
                v = levels[lv]["ph"][done]
                if not is_na(v):
                    lvl_latch[lv][0] = v
                v = levels[lv]["pl"][done]
                if not is_na(v):
                    lvl_latch[lv][1] = v
            lvl_series[lv]["h"][i] = lvl_latch[lv][0]
            lvl_series[lv]["l"][i] = lvl_latch[lv][1]

        a = atr_val[i]
        mod = float(p["htfAtrMod"])
        prox_sup = {lv: bool(p[f"{lv}On"]) and _is_near(c, lvl_latch[lv][1], a, mod) for lv in ("l1", "l2", "l3")}
        any_htf_resist = any(bool(p[f"{lv}On"]) and _is_near(c, lvl_latch[lv][0], a, mod) for lv in ("l1", "l2", "l3"))

        # SMC zones
        high_vol = (not p["useVol"]) or (not is_na(vol_sma[i]) and chart.volume[i] > vol_sma[i] * float(p["volMult"]))
        to_up = not is_na(zz_hi[i]) and h >= zz_hi[i]
        to_down = not is_na(zz_lo[i]) and l <= zz_lo[i]
        prev_zz = zz_trend
        if zz_trend == 1 and to_down:
            zz_trend = -1
        elif zz_trend == -1 and to_up:
            zz_trend = 1
        if exec_on["useSmc"]:
            if zz_trend == 1 and prev_zz == -1 and high_vol:
                ob = i
                lowest = l
                for j in range(0, min(21, i + 1)):
                    if chart.low[i - j] < lowest:
                        lowest = chart.low[i - j]
                        ob = i - j
                box = {"top": chart.high[ob], "bot": lowest, "left": ob, "kind": "demand", "born": i}
                smc_bu.append(box)
                smc_created.append(box)
                if len(smc_bu) > 5:
                    smc_bu.pop(0)
            if zz_trend == -1 and prev_zz == 1 and high_vol:
                ob = i
                highest = h
                for j in range(0, min(21, i + 1)):
                    if chart.high[i - j] > highest:
                        highest = chart.high[i - j]
                        ob = i - j
                box = {"top": highest, "bot": chart.low[ob], "left": ob, "kind": "supply", "born": i}
                smc_be.append(box)
                smc_created.append(box)
                if len(smc_be) > 5:
                    smc_be.pop(0)
        in_smc_demand = False
        in_smc_supply = False
        smc_demand_broken = False
        for box in smc_bu[:]:
            if l <= box["top"] and h >= box["bot"]:
                in_smc_demand = True
            if c < box["bot"]:
                smc_demand_broken = True
                box["died"] = i
                smc_bu.remove(box)
        for box in smc_be[:]:
            if h >= box["bot"] and l <= box["top"]:
                in_smc_supply = True
            if c > box["top"]:
                box["died"] = i
                smc_be.remove(box)

        # R triggers
        r_green = r_sw_l[i] or r_bu[i]
        r_red = r_sw_h[i] or r_be[i]

        # WVF signals
        w, wp = wvf[i], wvf_prev[i] if wvf_tf else (wvf[i - 1] if i > 0 else NAN)
        edge = wvf_edge[i]
        buy_sig = {}
        for tn in (1, 2, 3, 4, 5):
            buy_sig[tn] = edge and _crossover(w, wp, float(p[f"b{tn}Th"]))
        sell_sig = {}
        for tn in (1, 2, 3):
            sell_sig[tn] = edge and _crossunder(w, wp, float(p[f"s{tn}Th"]))
        z_prev = wvf_z[i - 1] if i > 0 else NAN
        pct_prev = wvf_pct[i - 1] if i > 0 else NAN
        stat_z_buy = bool(p["useZscore"]) and _crossover(wvf_z[i], z_prev, float(p["minZscore"]))
        stat_p_buy = bool(p["usePct"]) and _crossover(wvf_pct[i], pct_prev, float(p["minPct"]))
        is_fear_elevated = bool(p["blockHighWvf"]) and (not is_na(w)) and w > float(p["maxSellWvf"])

        equity = br.equity(c)
        # extreme fear unlocks the dry-powder reserve for every buy engine
        extreme = (not is_na(wvf_pct[i])) and wvf_pct[i] >= float(p["reserveFloorPct"])

        # -- gap zones: unfilled daily up-gaps (create at day open, kill on fill) --
        in_gap = False
        if p["gapBoostOn"]:
            kd = c2d_all[i]
            if day_edge[i] and kd >= 1 and \
                    daily.open[kd] > daily.close[kd - 1] * (1 + float(p["gapMinPct"]) / 100):
                gap_zones.append([daily.close[kd - 1], daily.open[kd]])
            if gap_zones:
                lo_i = chart.low[i]
                gap_zones = [z for z in gap_zones if lo_i > z[0]]
                in_gap = any(z[0] <= c <= z[1] for z in gap_zones)

        # -- A. passive accumulation grid --
        for tn in (1, 2, 3, 4, 5):
            if not (p[f"b{tn}En"] and buy_sig[tn]):
                continue
            if tn in (1, 2) and not can_buy_passive:
                continue
            qty = equity * (float(p[f"b{tn}Qty"]) / 100) / c
            br.queue_buy(f"Fear T{tn}", "grid", qty, f"Fear T{tn}", unlock_reserve=extreme)
            count(signal_counts, f"Fear T{tn}")

        # -- A2. favorability engine (lab extension): continuous sizing --
        if p["favOn"] and not is_na(wvf_pct[i]):
            floor = float(p["favFloorPct"])
            if p["gapBoostOn"] and in_gap and float(p["gapFloorDisc"]) > 0:
                floor = max(floor - float(p["gapFloorDisc"]), 0.0)
            gate = floor
            disc_ok = True
            if p["protectOn"] and prot_broken:
                gate = max(gate, float(p["protectFearFloor"]))   # broken regime: extreme fear only
                disc = float(p["protectRebuyDiscPct"])
                if disc > 0 and not is_na(prot_sale_px):         # ...and meaningfully below the sale
                    disc_ok = c < prot_sale_px * (1 - disc / 100)
            fav_cd = int(p["favCooldown"])
            if int(p["favCooldownExt"]) > 0 and wvf_pct[i] >= float(p["favCooldownExtFloor"]):
                fav_cd = int(p["favCooldownExt"])
            if disc_ok and wvf_pct[i] >= gate and (i - last_fav) >= fav_cd:
                f_base = (wvf_pct[i] - floor) / max(100.0 - floor, 1e-9)
                if p["favConfluence"]:
                    boost = 1.0
                    if (exec_on["useHtf"] and (prox_sup["l3"] or prox_sup["l2"] or prox_sup["l1"])) or \
                       (exec_on["useSmc"] and in_smc_demand):
                        boost += 0.5
                    if r_green:
                        boost += 0.5
                    f_base = min(f_base * boost, 1.0)
                if p["favDdOn"] and not is_na(dd_held[i]) and dd_held[i] > 0:
                    f_dd = min(dd_held[i] / (float(p["favDdRef"]) / 100), 1.0)
                    f_base = min(f_base * (1 + float(p["favDdX"]) * f_dd), 1.0)
                if float(p["favXFearBoost"]) > 0 and not is_na(xf_held[i]) \
                        and xf_held[i] >= float(p["favXFearFloor"]):
                    f_base = min(f_base * float(p["favXFearBoost"]), 1.0)
                    count(signal_counts, "FavXFear")
                if div_cnt is not None and div_cnt[i] > 0:
                    f_base = min(f_base * (1 + float(p["favDivX"]) * div_cnt[i]), 1.0)
                    count(signal_counts, "FavDiv")
                if p["favNweOn"] and not is_na(z_held[i]) \
                        and z_held[i] <= -float(p["favNweFloor"]):
                    exc = -z_held[i] - float(p["favNweFloor"])
                    f_base = min(f_base * (1 + float(p["favNweX"]) * (1 + exc)), 1.0)
                    count(signal_counts, "FavNwe")
                if nl_held is not None and not is_na(nl_held[i]) \
                        and nl_held[i] >= float(p["favNlFloor"]):
                    f_base = min(f_base * float(p["favNlX"]), 1.0)
                    count(signal_counts, "FavNl")
                if strec_held is not None and strec_held[i]:
                    f_base = min(f_base * float(p["favStX"]), 1.0)
                    count(signal_counts, "FavStRec")
                pct = float(p["favMinPct"]) + (float(p["favMaxPct"]) - float(p["favMinPct"])) * (f_base ** float(p["favPower"]))
                if p["gapBoostOn"] and in_gap:
                    boosted = min(pct * float(p["gapBoostX"]), float(p["favMaxPct"]))
                    if boosted > pct:
                        count(signal_counts, "FavGap")
                    pct = boosted
                fav_qty = equity * (pct / 100) / c
                if fav_atr is not None and not is_na(fav_atr[i]) and in_win:
                    # limit-bid: rest one bar below the close, market fallback
                    br.set_limit("Fav", "grid",
                                 c - float(p["favLimitPadAtr"]) * fav_atr[i],
                                 fav_qty, "Fav")
                    fav_limit = {"bar": i, "qty": fav_qty, "unlock": extreme}
                else:
                    br.queue_buy("Fav", "grid", fav_qty, "Fav", unlock_reserve=extreme)
                count(signal_counts, "Fav")
                last_fav = i
                if p["protectOn"] and prot_broken:
                    protect_pool = max(0.0, protect_pool - equity * (pct / 100))
                if glide_n > 0:          # fear buys consume parcels oldest-first
                    spend = equity * (pct / 100)
                    for parc in glide_parcels:
                        take = min(parc[1], spend)
                        parc[1] -= take
                        spend -= take
                        if spend <= 0:
                            break
                    glide_parcels = [parc for parc in glide_parcels if parc[1] > 1.0]

        # -- B. statistical accumulation --
        if stat_z_buy:
            br.queue_buy("Stat Z", "sniper", equity * (float(p["zQty"]) / 100) / c, "Stat Z",
                         unlock_reserve=extreme)
            count(signal_counts, "Stat Z")
        if stat_p_buy:
            br.queue_buy("Stat Pct", "sniper", equity * (float(p["pQty"]) / 100) / c, "Stat Pct",
                         unlock_reserve=extreme)
            count(signal_counts, "Stat Pct")

        # -- C. snipers --
        sl_stop = (c * (1 - float(p["slHardPct"]) / 100)) if p["useSlHard"] else None
        level_sniper = False
        is_at_level = (exec_on["useHtf"] and (prox_sup["l3"] or prox_sup["l2"] or prox_sup["l1"])) or \
                      (exec_on["useSmc"] and in_smc_demand)
        if p["useRIn"] and r_green:
            if is_at_level and use_conf:
                if exec_on["useHtf"] and prox_sup["l3"]:
                    br.queue_buy("L3 Sniper", "sniper", equity * (float(p["l3Qty"]) / 100) / c, "L3 Sniper", sl_stop, extreme)
                    count(signal_counts, "L3 Sniper")
                    level_sniper = True
                elif exec_on["useHtf"] and prox_sup["l2"]:
                    br.queue_buy("L2 Sniper", "sniper", equity * (float(p["l2Qty"]) / 100) / c, "L2 Sniper", sl_stop, extreme)
                    count(signal_counts, "L2 Sniper")
                    level_sniper = True
                elif exec_on["useSmc"] and in_smc_demand:
                    br.queue_buy("SMC Sniper", "sniper", equity * (float(p["smcQty"]) / 100) / c, "SMC Sniper", sl_stop, extreme)
                    count(signal_counts, "SMC Sniper")
                    level_sniper = True
                elif exec_on["useHtf"] and prox_sup["l1"]:
                    br.queue_buy("L1 Sniper", "sniper", equity * (float(p["l1Qty"]) / 100) / c, "L1 Sniper", sl_stop, extreme)
                    count(signal_counts, "L1 Sniper")
                    level_sniper = True
            if (not is_na(w)) and w > float(p["oppWvfMin"]) and not level_sniper:
                br.queue_buy("Opp Sniper", "sniper", equity * (float(p["rInQty"]) / 100) / c, "Opp Sniper", sl_stop, extreme)
                count(signal_counts, "Opp Sniper")

        # -- limit-bid mode (continuous) --
        if use_limit and exec_on["useHtf"] and not is_na(a):
            pad = a * float(p["limitPad"])
            for lv, eng in (("l3", "L3 Limit"), ("l2", "L2 Limit"), ("l1", "L1 Limit")):
                if not p[f"{lv}On"]:
                    continue
                ll = lvl_latch[lv][1]
                if not is_na(ll) and c < ll + a * 5:
                    px = ll + pad
                    stop = px * (1 - float(p["slHardPct"]) / 100) if p["useSlHard"] else None
                    br.set_limit(eng, "sniper", px, equity * (float(p[f"{lv}Qty"]) / 100) / px, eng, stop)

        # -- thesis exits --
        if p["useSlThesis"]:
            if smc_demand_broken:
                for lot in [x for x in br.lots if x.engine == "SMC Sniper"]:
                    br.queue_sell_lot(lot, 1.0, "Zone Failed")
                    count(sell_counts, "Zone Failed")
            for lv, eng in (("l3", "L3 Sniper"), ("l2", "L2 Sniper"), ("l1", "L1 Sniper")):
                ll = lvl_latch[lv][1]
                if not is_na(ll) and not is_na(a) and c < ll - a * 0.5:
                    for lot in [x for x in br.lots if x.engine == eng]:
                        br.queue_sell_lot(lot, 1.0, f"{eng[:2]} Failed")
                        count(sell_counts, f"{eng[:2]} Failed")
                    br.cancel_limit(eng.replace("Sniper", "Limit"))

        # -- D. distribution --
        if br.lots:
            sig_s = {tn: p[f"s{tn}En"] and sell_sig[tn] for tn in (1, 2, 3)}
            sig_tech = bool(p["useROut"]) and r_red and (in_smc_supply or any_htf_resist)
            best_lot = None
            best_pnl = -9999.0
            for lot in br.lots[:]:
                pnl_pct = (c - lot.entry_price) / lot.entry_price * 100
                is_sniper = lot.cls == "sniper"
                req_check = p["snipOnlyProfit"] if is_sniper else p["gridOnlyProfit"]
                req_min = float(p["snipMinProfit"]) if is_sniper else float(p["gridMinProfit"])
                meets = (not req_check) or (pnl_pct >= req_min)
                if pnl_pct > best_pnl:
                    best_pnl = pnl_pct
                    best_lot = lot
                if not meets or is_fear_elevated:
                    continue
                if sig_s[1]:
                    br.queue_sell_lot(lot, float(p["s1Qty"]) / 100, "Greed 1")
                    count(sell_counts, "Greed 1")
                elif sig_s[2]:
                    br.queue_sell_lot(lot, float(p["s2Qty"]) / 100, "Greed 2")
                    count(sell_counts, "Greed 2")
                elif sig_s[3]:
                    br.queue_sell_lot(lot, float(p["s3Qty"]) / 100, "Greed 3")
                    count(sell_counts, "Greed 3")
                elif sig_tech and not p["techExitBest"]:
                    allowed = (pnl_pct > 0) if p["techIgnMin"] else meets
                    if allowed:
                        br.queue_sell_lot(lot, float(p["rOutQty"]) / 100, "Tech Exit")
                        count(sell_counts, "Tech Exit")
            if sig_tech and p["techExitBest"] and best_lot is not None and not is_fear_elevated:
                is_sniper = best_lot.cls == "sniper"
                req_check = p["snipOnlyProfit"] if is_sniper else p["gridOnlyProfit"]
                req_min = float(p["snipMinProfit"]) if is_sniper else float(p["gridMinProfit"])
                meets = (not req_check) or (best_pnl >= req_min)
                allowed = (best_pnl > 0) if p["techIgnMin"] else meets
                if allowed:
                    br.queue_sell_lot(best_lot, float(p["rOutQty"]) / 100, "Tech Exit Best")
                    count(sell_counts, "Tech Exit Best")

        # -- E. tiered scale-out ladder (lab extension): daily z, gains only --
        if day_edge[i]:
            day_count += 1
        def do_trim(eq_pct: float, tag: str) -> float:     # ladder + vol-blowoff + omen
            target = equity * eq_pct / 100
            sold = 0.0
            min_trim = float(p.get("minTrimUsd", 0.0))
            pool = [lot for lot in br.lots
                    if (c / lot.entry_price - 1) * 100 >= float(p["zSellMinGainPct"])]
            order = p["trimLotOrder"]
            if order == "skipArmed":      # spare the trail engine's armed cohort
                eligible = sorted((lot for lot in pool if lot.serial not in lot_peak),
                                  key=lambda lo: lo.entry_price)
            elif order == "minGain":      # smallest gain above the floor first
                eligible = sorted(pool, key=lambda lo: lo.entry_price, reverse=True)
            elif order == "oldest":       # entry order
                eligible = sorted(pool, key=lambda lo: lo.entry_ts)
            else:                         # "best": best P&L (lowest cost) first
                eligible = sorted(pool, key=lambda lo: lo.entry_price)
            for lot in eligible:
                if sold >= target:
                    break
                lot_val = lot.qty * c
                frac = min(1.0, (target - sold) / lot_val) if lot_val > 0 else 0.0
                if frac <= 0:
                    continue
                if lot_val * frac < min_trim:   # skip sub-floor dust slices
                    continue
                br.queue_sell_lot(lot, frac, tag)
                sold += lot_val * frac
            return sold

        if p["zSellOn"] and day_edge[i] and not is_na(z_held[i]):
            z = z_held[i]
            er_gate = float(p["zSellErMax"]) > 0 and not is_na(er_held[i]) \
                and er_held[i] > float(p["zSellErMax"])
            for tn in (1, 2, 3):
                if not p[f"zt{tn}En"]:
                    continue
                th = float(p[f"zt{tn}Z"])
                if zt_armed[tn]:
                    if z >= th and br.lots and (day_count - zt_last[tn]) >= int(p["zSellCooldownDays"]) \
                            and not (er_gate and tn < 3):
                        tag = f"Trim z{th:g}"
                        sold = do_trim(float(p[f"zt{tn}Pct"]), tag)
                        if sold > 0:
                            count(sell_counts, tag)
                            zt_armed[tn] = False
                            zt_last[tn] = day_count
                            recycle_pool += sold
                elif z < th - float(p["zSellRearm"]):
                    zt_armed[tn] = True

        # -- E1b. volume-climax blowoff sell (EXP-B): never ER-gated --
        if blow_held is not None and day_edge[i] and blow_held[i] and br.lots \
                and (day_count - vb_last) >= int(p["volBlowCooldownDays"]):
            sold = do_trim(float(p["volBlowPct"]), "VolBlowoff")
            if sold > 0:
                count(sell_counts, "VolBlowoff")
                recycle_pool += sold
                vb_last = day_count

        # -- E1c. Hindenburg-omen trim (EXP-C): bypasses the ER gate --
        if hind_held is not None and day_edge[i] and hind_held[i] and br.lots \
                and (day_count - hd_last) >= int(p["hindSellCooldownDays"]):
            sold = do_trim(float(p["hindSellPct"]), "Hindenburg")
            if sold > 0:
                count(sell_counts, "Hindenburg")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                hd_last = day_count
                exo_fire_day = day_count

        # -- E1d. credit-deterioration trim (IF1-2): bypasses the ER gate --
        if cred_held is not None and day_edge[i] and cred_held[i] and br.lots \
                and (day_count - cr_last) >= int(p["credSellCooldownDays"]):
            cred_pct = float(p["credSellPct"])
            if float(p["credSevSlope"]) > 0:
                cred_pct = min(float(p["credSevMaxPct"]),
                               cred_pct + float(p["credSevSlope"]) * cred_sev_held[i])
            sold = do_trim(cred_pct, "CreditTrim")
            if sold > 0:
                count(sell_counts, "CreditTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                cr_last = day_count
                exo_fire_day = day_count

        # -- E1e. participation-breadth trim (ratio sweep): ER-gate bypass --
        if pb_held is not None and day_edge[i] and pb_held[i] and br.lots \
                and (day_count - pb_last) >= int(p["pbSellCooldownDays"]):
            sold = do_trim(float(p["pbSellPct"]), "PartTrim")
            if sold > 0:
                count(sell_counts, "PartTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                pb_last = day_count
                exo_fire_day = day_count

        # -- E1f. dollar-shock trim (IF3 if3hs): bypasses the ER gate --
        if dxy_held is not None and day_edge[i] and dxy_held[i] and br.lots \
                and (day_count - dx_last) >= int(p["dxySellCooldownDays"]):
            sold = do_trim(float(p["dxySellPct"]), "DollarTrim")
            if sold > 0:
                count(sell_counts, "DollarTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                dx_last = day_count
                exo_fire_day = day_count

        # -- E1g. semis-leadership trim (IF4 if4fb): bypasses the ER gate --
        if sox_held is not None and day_edge[i] and sox_held[i] and br.lots \
                and (day_count - sx_last) >= int(p["soxSellCooldownDays"]):
            sold = do_trim(float(p["soxSellPct"]), "SoxTrim")
            if sold > 0:
                count(sell_counts, "SoxTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                sx_last = day_count
                exo_fire_day = day_count

        # -- E1h. transports-leadership trim (IF5 if5nx): ER-gate bypass --
        if djt_held is not None and day_edge[i] and djt_held[i] and br.lots \
                and (day_count - dj_last) >= int(p["djtSellCooldownDays"]):
            sold = do_trim(float(p["djtSellPct"]), "TransTrim")
            if sold > 0:
                count(sell_counts, "TransTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                dj_last = day_count
                exo_fire_day = day_count

        # -- E1i. inflation-expectations trim (IF7 if7rg): ER-gate bypass --
        if tip_held is not None and day_edge[i] and tip_held[i] and br.lots \
                and (day_count - tp_last) >= int(p["tipSellCooldownDays"]):
            sold = do_trim(float(p["tipSellPct"]), "TipTrim")
            if sold > 0:
                count(sell_counts, "TipTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                tp_last = day_count
                exo_fire_day = day_count

        # -- E1j. speculative-appetite trim (IF8 if8sb): ER-gate bypass --
        if ipo_held is not None and day_edge[i] and ipo_held[i] and br.lots \
                and (day_count - ip_last) >= int(p["ipoSellCooldownDays"]):
            sold = do_trim(float(p["ipoSellPct"]), "IpoTrim")
            if sold > 0:
                count(sell_counts, "IpoTrim")
                if not p["exoProceedsToCash"]:
                    recycle_pool += sold
                ip_last = day_count
                exo_fire_day = day_count

        # -- E2. per-lot trailing gain harvest (lab extension) --
        trail_er_block = float(p["lotTrailErMax"]) > 0 and not is_na(er_held[i]) \
            and er_held[i] > float(p["lotTrailErMax"])
        if p["lotTrailOn"] and br.lots:
            _in_exo = (int(p["lotTrailExoWindowDays"]) > 0 and
                       (day_count - exo_fire_day) < int(p["lotTrailExoWindowDays"]))
            arm, retr, tfrac = _eff_trail(p, trail_atr_held[i], z_held[i],
                                          er_held[i], _in_exo)
            min_gain = float(p["zSellMinGainPct"]) / 100
            ef_min = float(p["lotTrailEntryFMin"])
            for lot in br.lots:
                gain = c / lot.entry_price - 1
                pk = lot_peak.get(lot.serial)
                if pk is None:
                    if gain >= arm:
                        if ef_min > 0:
                            ef = wvf_pct[lot.entry_idx] if 0 <= lot.entry_idx < n else NAN
                            if is_na(ef) or ef < ef_min:
                                continue             # not a fear-born lot: never trails
                        lot_peak[lot.serial] = c     # armed: start trailing
                    continue
                if c > pk:
                    lot_peak[lot.serial] = c
                elif c <= pk * (1 - retr) and gain >= min_gain and not trail_er_block \
                        and (i - lot_fire.get(lot.serial, -(10 ** 9))) >= int(p["lotTrailCooldown"]):
                    br.queue_sell_lot(lot, tfrac, "Trail")
                    count(sell_counts, "Trail")
                    recycle_pool += lot.qty * tfrac * c
                    lot_fire[lot.serial] = i
                    lot_peak[lot.serial] = c         # reset peak: no cascade refire

        # -- F. redeploy trim proceeds at levels (lab extension) --
        redeploy_fear = float(p["redeployFearFloor"]) > 0 and not is_na(wvf_pct[i]) \
            and wvf_pct[i] >= float(p["redeployFearFloor"])
        redeploy_ob = rsiob_held is not None and rsiob_held[i]
        exo_blocked = int(p["redeployExoBlockDays"]) > 0 and \
            (day_count - exo_fire_day) < int(p["redeployExoBlockDays"])
        if p["redeployOn"] and (is_at_level or redeploy_fear or redeploy_ob) and recycle_pool > 1.0 \
                and not exo_blocked \
                and (i - last_redeploy) >= int(p["redeployCooldown"]):
            usd = min(recycle_pool, br.cash) * float(p["redeployPct"]) / 100
            if usd > 1.0:
                br.queue_buy("Redeploy", "grid", usd / c, "Redeploy", unlock_reserve=extreme)
                count(signal_counts, "Redeploy")
                recycle_pool -= usd
                last_redeploy = i

        # -- G. protect/de-risk layer (lab extension): daily SMA regime gate --
        if p["protectOn"] and day_edge[i] and not is_na(prot_sma_held[i]):
            dc, sma = prot_close_held[i], prot_sma_held[i]
            if not prot_broken and dc < sma * (1 - float(p["protectBreakPct"]) / 100):
                prot_broken = True
                prot_break_i = i
                prot_sale_px = dc
                frac = float(p["protectSellPct"]) / 100
                sold = 0.0
                for lot in br.lots:
                    br.queue_sell_lot(lot, frac, "Protect")
                    sold += lot.qty * frac * c
                if sold > 0:
                    count(sell_counts, "Protect")
                    protect_pool += sold
            elif prot_broken and dc > sma * (1 + float(p["protectReclaimPct"]) / 100):
                prot_broken = False
                prot_reclaim_i = i
                if protect_pool > 1.0:
                    br.queue_buy("Reclaim", "grid", protect_pool / c, "Reclaim", unlock_reserve=True)
                    count(signal_counts, "Reclaim")
                protect_pool = 0.0

        # -- window gate: warmup/last bars run every signal but trade nothing --
        if not in_win:
            if i == n - 1:
                # DECISION BAR (li): capture the orders the engines queued on the last
                # closed bar BEFORE this gate wipes them. li has NO next bar, so these
                # never fill -> the backtest board is byte-identical; but they ARE the
                # live signal (what the strategy would submit MOO for the next real
                # open). Without this the collect_* exports read an empty post-loop
                # pending_buys (earlier bars' orders were consumed by on_bar_open(i+1);
                # li's are wiped right here) -> grid/snipers/redeploy would SILENTLY
                # never emit a live order, and fired_engines (the coverage guard input)
                # would be empty so the guard could never catch it.
                decision_pending_buys = list(br.pending_buys)
                decision_pending_sells = list(br.pending_sells)
                decision_working_limits = dict(br.working_limits)
            br.pending_buys.clear()
            br.pending_sells.clear()
            br.working_limits.clear()

        # -- bear hedge: while broken, sweep idle cash into the inverse ETF (so the
        # position PROFITS from the decline); on reclaim, liquidate it back to cash.
        # Marked at the hedge close -> the ETF's real expense/decay are included. The
        # Reclaim/normal buys redeploy from the liquidated cash (broker clips to it). --
        if hedge_px is not None and not is_na(hedge_px[i]) and w0 <= i:
            hpx = hedge_px[i]
            if prot_broken:
                if br.cash > 1.0:
                    hedge_shares += br.cash / hpx
                    br.cash = 0.0
            elif hedge_shares > 0:
                br.cash += hedge_shares * hpx
                hedge_shares = 0.0
        hedge_val = hedge_shares * hedge_px[i] if (hedge_px is not None and not is_na(hedge_px[i])) else 0.0

        # -- mark equity --
        q = br.position_qty()
        eq_close[i] = br.cash + q * c + hedge_val
        eq_high[i] = br.cash + q * h + hedge_val
        eq_low[i] = br.cash + q * l + hedge_val
        pos_frac[i] = (q * c) / eq_close[i] if eq_close[i] > 0 else 0.0
        bench_eq[i] = bench_qty * c

    # ---------------------------------------------------------------- metrics
    total_in = sum(amt for _, amt in br.flows)
    final_eq = eq_close[w1]
    bench_final = bench_eq[w1]
    strat_irr = xirr(br.flows, chart.ts[w1], final_eq)
    bench_irr = xirr(br.flows, chart.ts[w1], bench_final)
    mdd = max_drawdown(eq_high[w0:w1 + 1], eq_low[w0:w1 + 1])
    bench_mdd = max_drawdown(bench_eq[w0:w1 + 1], bench_eq[w0:w1 + 1])

    attribution: dict[str, dict] = {}
    for f in br.fills:
        e = attribution.setdefault(f["engine"], {"buys": 0, "usd_in": 0.0, "realized": 0.0, "open_pnl": 0.0, "open_usd": 0.0})
        e["buys"] += 1
        e["usd_in"] += f["usd"]
    for t in br.trades:
        attribution.setdefault(t["engine"], {"buys": 0, "usd_in": 0.0, "realized": 0.0, "open_pnl": 0.0, "open_usd": 0.0})
        attribution[t["engine"]]["realized"] += t["pnl"]
    last_c = chart.close[w1]
    for lot in br.lots:
        e = attribution.setdefault(lot.engine, {"buys": 0, "usd_in": 0.0, "realized": 0.0, "open_pnl": 0.0, "open_usd": 0.0})
        e["open_pnl"] += (last_c - lot.entry_price) * lot.qty
        e["open_usd"] += lot.qty * last_c

    wins = sum(1 for t in br.trades if t["pnl"] > 0)
    gross_win = sum(t["pnl"] for t in br.trades if t["pnl"] > 0)
    gross_loss = abs(sum(t["pnl"] for t in br.trades if t["pnl"] < 0))
    # TV-style profit factor: realized trades only, gross wins / gross losses
    profit_factor = round(gross_win / gross_loss, 2) if gross_loss > 0 else None
    peak = -math.inf
    mdd_usd = 0.0
    for i in range(w0, w1 + 1):
        if eq_high[i] > peak:
            peak = eq_high[i]
        if peak - eq_low[i] > mdd_usd:
            mdd_usd = peak - eq_low[i]
    wn = w1 - w0 + 1
    # C5: realistic per-order cost that the modeled %-of-notional commission
    # misses. IBKR Pro US-equity schedule: $0.0035/share, min $0.35/order,
    # capped at 1% of trade value. Reported ONLY (does not touch sim equity) so
    # the per-order floor a pure-bps model hides on tiny fills is visible — with
    # the board's default commissionPct=0, spend == floor x fill_count. The
    # churn driver is SMALL sell-frac (many tiny fills), not 100% harvest. See
    # LEDGER "C5 realistic cost".
    def _ibkr_spend(qty, notional):
        return min(max(0.0035 * qty, 0.35), 0.01 * notional)
    spend = sum(_ibkr_spend(f["qty"], f["usd"]) for f in br.fills)
    spend += sum(_ibkr_spend(t["qty"], t["qty"] * t["exit"]) for t in br.trades)
    fills_total = len(br.fills) + len(br.trades)
    metrics = {
        "bars": wn,
        "start": time.strftime("%Y-%m-%d", time.gmtime(chart.ts[w0])),
        "end": time.strftime("%Y-%m-%d", time.gmtime(chart.ts[w1])),
        "money_in": round(total_in, 2),
        "final_equity": round(final_eq, 2),
        "net_pnl": round(final_eq - total_in, 2),
        "net_pct_on_money_in": round((final_eq / total_in - 1) * 100, 2),
        "irr_pct": round(strat_irr * 100, 2) if not is_na(strat_irr) else None,
        "max_dd_pct": round(mdd, 2),
        "max_dd_usd": round(mdd_usd, 2),
        "profit_factor": profit_factor,
        "gross_win": round(gross_win, 2),
        "gross_loss": round(gross_loss, 2),
        "bench_final_equity": round(bench_final, 2),
        "bench_net_pct_on_money_in": round((bench_final / total_in - 1) * 100, 2) if total_in > 0 else None,
        "bench_irr_pct": round(bench_irr * 100, 2) if not is_na(bench_irr) else None,
        "bench_max_dd_pct": round(bench_mdd, 2),
        # REMOVED 2026-06-14 (user: not a metric they evaluate on): the wealth-ratio
        # `edge_vs_bench_pct` = (final_eq / bench_final - 1) * 100. It ranked configs
        # differently from the board edge and was the source of a recurring metric
        # trap. The ONLY edge metric is the BOARD edge = net_pct_on_money_in -
        # bench_net_pct_on_money_in (pp), computed by consumers. Do not re-add it.
        "exposure_avg_pct": round(sum(pos_frac[w0:w1 + 1]) / wn * 100, 1),
        "exposure_end_pct": round(pos_frac[w1] * 100, 1),
        "cash_end": round(br.cash, 2),
        "buy_fills": len(br.fills),
        "sell_fills": len(br.trades),
        "realized_pnl": round(br.realized_pnl(), 2),
        "realized_win_rate_pct": round(wins / len(br.trades) * 100, 1) if br.trades else None,
        "buys_clipped": br.buys_clipped,
        "buys_skipped": br.buys_skipped,
        "open_lots": len(br.lots),
        "spend_ibkr": round(spend, 2),
        "spend_pct_of_money_in": round(spend / total_in * 100, 3) if total_in > 0 else None,
        "spend_per_fill": round(spend / fills_total, 4) if fills_total else None,
        "fills_total": fills_total,
    }
    open_lots = sorted(({"engine": lot.engine, "ts": lot.entry_ts, "qty": round(lot.qty, 4),
                         "entry": round(lot.entry_price, 2),
                         "pnl": round((last_c - lot.entry_price) * lot.qty, 2),
                         "pnl_pct": round((last_c / lot.entry_price - 1) * 100, 2)}
                        for lot in br.lots), key=lambda x: x["ts"])

    # ------------------------------------------- live decision snapshot
    # Read-only export of the engine's end-of-data state for the --signal
    # decision sheet (and, later, the IBKR bridge). Computed AFTER the bar
    # loop from the same arrays the sim traded on — zero behavior change.
    # "Tomorrow" = the next chart day: its bars read daily index k-1 =
    # the LAST confirmed daily bar (jl), so pending regime trims are
    # fully determined tonight; intraday engines report trigger levels.
    jl = len(daily) - 1
    li = n - 1
    day_tomorrow = day_count + 1

    # The sells the sim actually QUEUED on the decision bar li (from the snapshot,
    # captured before the window gate wiped it). Every repeating sell export below
    # gates its fire flag on THIS -- a tag is "live" IFF the sim queued it on li --
    # instead of a current-bar predicate + a cooldown vs the frozen sim *_last. That
    # fixes TWO Stage-1-surfaced sell-side divergences: (1) the HELD-SHIFT (the export
    # led the sim by a bar: it read the RAW d_<x>[jl] while the sim fires off the HELD
    # predicate cred_held[i]=d_<x>[i-1]); (2) the OVER-EMISSION (the export re-fired
    # every bar of a persistent omen because it could not replicate the sim's br.lots /
    # sold>0 gate, so its cooldown stayed elapsed). Faithful-by-construction, exactly
    # like the snapshot-derived buys. The per-tag DOSE stays param-based (the bridge
    # applies it to its OWN ledger lots via sizing.resolve). Protect/reclaim (one-shot
    # regime transitions, no repeat/cooldown) are unaffected -- they already align.
    fired_sell_tags = {(o.get("tag") or "") for o in decision_pending_sells}

    def _trim_live(tag, last_fire, cd_days, dose_pct):
        return {"tag": tag, "signal": tag in fired_sell_tags,
                "cooldown_ok": (day_tomorrow - last_fire) >= cd_days,     # informational
                "cooldown_left_days": max(0, cd_days - (day_tomorrow - last_fire)),
                "dose_pct": round(dose_pct, 1)}

    live_trims = []
    if hind_held is not None:
        live_trims.append(_trim_live(
            "Hindenburg", hd_last,
            int(p["hindSellCooldownDays"]), float(p["hindSellPct"])))
    if cred_held is not None:
        dose = float(p["credSellPct"])
        if float(p["credSevSlope"]) > 0:
            # cred_sev_held[li] (= d_sev[jl-1]), NOT the raw d_sev[jl]: the sim's
            # fire on li sizes off the HELD severity (do_trim, sec E1d), so a
            # snapshot-gated CreditTrim must carry the dose the sim executed --
            # the raw read led it by one daily bar (5pp inside the 80..85 band).
            dose = min(float(p["credSevMaxPct"]),
                       dose + float(p["credSevSlope"]) * cred_sev_held[li])
        live_trims.append(_trim_live(
            "CreditTrim", cr_last,
            int(p["credSellCooldownDays"]), dose))
    if pb_held is not None:
        live_trims.append(_trim_live(
            "PartTrim", pb_last,
            int(p["pbSellCooldownDays"]), float(p["pbSellPct"])))
    if dxy_held is not None:
        live_trims.append(_trim_live(
            "DollarTrim", dx_last,
            int(p["dxySellCooldownDays"]), float(p["dxySellPct"])))
    if sox_held is not None:
        live_trims.append(_trim_live(
            "SoxTrim", sx_last,
            int(p["soxSellCooldownDays"]), float(p["soxSellPct"])))
    if djt_held is not None:
        live_trims.append(_trim_live(
            "TransTrim", dj_last,
            int(p["djtSellCooldownDays"]), float(p["djtSellPct"])))
    if tip_held is not None:
        live_trims.append(_trim_live(
            "TipTrim", tp_last,
            int(p["tipSellCooldownDays"]), float(p["tipSellPct"])))
    if ipo_held is not None:
        live_trims.append(_trim_live(
            "IpoTrim", ip_last,
            int(p["ipoSellCooldownDays"]), float(p["ipoSellPct"])))

    arm_f = float(p["lotTrailArmPct"]) / 100
    retr_f = float(p["lotTrailRetracePct"]) / 100
    live_lots = []
    for lot in sorted(br.lots, key=lambda x: x.entry_ts):
        pk = lot_peak.get(lot.serial)
        live_lots.append({
            "engine": lot.engine, "ts": lot.entry_ts, "qty": round(lot.qty, 4),
            "entry": round(lot.entry_price, 2),
            "gain_pct": round((last_c / lot.entry_price - 1) * 100, 2),
            "trail_armed": pk is not None,
            "trail_trigger": round(pk * (1 - retr_f), 2) if pk is not None else None,
            "arm_price": None if pk is not None
            else round(lot.entry_price * (1 + arm_f), 2),
        })

    fear_pct = wvf_pct[li]
    size_pct = None
    if not is_na(fear_pct) and fear_pct >= float(p["favFloorPct"]):
        fb = (fear_pct - float(p["favFloorPct"])) / max(100.0 - float(p["favFloorPct"]), 1e-9)
        size_pct = round(float(p["favMinPct"]) + (float(p["favMaxPct"]) - float(p["favMinPct"]))
                         * (fb ** float(p["favPower"])), 2)
    n_block = int(p["redeployExoBlockDays"])
    blk_left = max(0, n_block - (day_tomorrow - exo_fire_day)) if n_block > 0 else 0
    live = {
        "last_bar_ts": chart.ts[li],
        "last_daily_ts": daily.ts[jl],
        "last_close": round(last_c, 2),
        "equity": round(br.cash + br.position_qty() * last_c
                        + (hedge_shares * hedge_px[li] if (hedge_px is not None and not is_na(hedge_px[li])) else 0.0), 2),
        "cash": round(br.cash, 2),
        "recycle_pool": round(recycle_pool, 2),
        "open_lots": len(br.lots),
        "profitable_lots": sum(1 for x in br.lots if last_c > x.entry_price),
        "er": None if is_na(er_held[li]) else round(er_held[li], 3),
        "er_harvest_max": float(p["zSellErMax"]),
        "z": None if is_na(z_held[li]) else round(z_held[li], 2),
        "fear": {
            "wvf_pctile": None if is_na(fear_pct) else round(fear_pct, 1),
            "floor": float(p["favFloorPct"]),
            "cooldown_bars_left": max(0, int(p["favCooldown"]) - (li - last_fav)),
            "size_pct_if_fired": size_pct,
            # fired_pct: the EXACT executed fav pct recovered from the decision-bar
            # snapshot (qty*last_close/equity*100, boosts included), or None if the sim
            # did NOT queue Fav on li. This is the faithful emit trigger + size -- the
            # bridge emits fav off THIS (not size_pct_if_fired / cooldown_bars_left, which
            # omit boosts and invert on the fire bar). See collect_fav_buy + intent.py.
            "fired_pct": collect_fav_buy(decision_pending_buys, last_c, br.equity(last_c)),
        },
        "trims": live_trims,
        "exo_block": {"days": n_block, "active": blk_left > 0,
                      "remaining_days": blk_left},
        "redeploy": {
            "pool": round(recycle_pool, 2),
            # fire_usd: the recycle-pool redeploy buy QUEUED on the decision bar
            # (qty * last_close), derived faithfully from the DECISION-BAR SNAPSHOT
            # exactly like the grid/snipers (queue_buy -> fills next open = MOO on a 1d
            # cell). 0.0 = no redeploy queued on li. Read-only -> board byte-identical.
            # (A post-loop br.pending_buys read would be EMPTY -- li wiped by the gate.)
            # The bridge emits this as a kind="usd" MOO on 1d cells (intraday cells keep
            # redeploy deferred). See collect_redeploy_buy + fearlab/bridge/intent.py.
            "fire_usd": collect_redeploy_buy(decision_pending_buys, last_c),
            "cooldown_bars_left": max(0, int(p["redeployCooldown"]) - (li - last_redeploy)),
            "levels": {k: {"h": (None if is_na(lvl_series[k]["h"][li])
                                 else round(lvl_series[k]["h"][li], 2)),
                           "l": (None if is_na(lvl_series[k]["l"][li])
                                 else round(lvl_series[k]["l"][li], 2))}
                       for k in ("l1", "l2", "l3")},
        },
        "lots": live_lots,
    }

    # ---- engine-coverage export: the EXACT set of engines that fired on the
    # decision bar (li), collected post-loop from the still-pending broker orders.
    # Read-only -> the simulate loop is untouched and the board is byte-identical.
    # Reads the DECISION-BAR snapshot (orders queued on li, captured before the window
    # gate wiped them) -- NOT the post-loop br.pending_buys, which is always empty (li
    # is wiped; earlier bars consumed by on_bar_open). See collect_fired_engines.
    live["fired_engines"] = collect_fired_engines(
        decision_pending_buys, decision_pending_sells, decision_working_limits)

    # ---- passive grid export: the accumulation grid (tiers 1..5) fires in-sim as
    # br.queue_buy(f"Fear T{n}", "grid", qty=equity*b{n}Qty/100/c, ...). The orders
    # queued on the decision bar (li) fill at next open (= MOO on a 1d cell, same fill
    # class as the Fav fear buy). We derive the live grid intents FAITHFULLY from the
    # DECISION-BAR SNAPSHOT (li's queued orders, captured in the window gate before the
    # wipe): a tier appears here iff the sim queued it on li -> every sim gate (b{n}En,

    # simulate loop is untouched -> board byte-identical. NB a POST-loop br.pending_buys
    # read would be EMPTY (li wiped by the gate) -> this export must use the snapshot.
    # See the module-level collect_grid_buys for the full rationale.

    # -- derived faithfully from the DECISION-BAR SNAPSHOT exactly like the grid
    # (queue_buy -> fills next open = MOO on a 1d cell). Read-only -> board byte-
    # identical. See collect_sniper_buys for the full rationale.
    live["snipers"] = collect_sniper_buys(decision_pending_buys, p)

    # live snapshot — the protect/reclaim daily-SMA regime gate (sec G), the
    # z-ladder scale-out (sec E), and the volume-climax blowoff (sec E1b). All
    # three are daily-confirmed (fire on day_edge, known overnight) yet were
    # never exported, so the execution bridge would silently skip the strategy's
    # protective sells. Read-only, computed from post-loop state; zero behavior
    # change (no sim variable is mutated here). See fearlab/bridge/intent.py.
    prot_dc, prot_sma = prot_close_held[li], prot_sma_held[li]
    _has_sma = not is_na(prot_sma)
    # bear-hedge live snapshot: the inverse-ETF leg (protectBearHedge, e.g. PSQ) is
    # simulated as a scalar `hedge_shares` swept inside the loop and folded into the
    # `equity` above -> bridge/intent.py was blind to it, so a live bear cell would
    # trade the long sleeve and SILENTLY DROP the hedge. Export ticker/px/shares/active
    # here (+ a standalone hedge_mark so sizing can de-fold it from the long-only
    # equity). Read-only; zero sim behavior change. See fearlab/bridge/intent.py.
    _hpx = hedge_px[li] if (hedge_px is not None and not is_na(hedge_px[li])) else None
    live["protect"] = {
        "on": bool(p["protectOn"]),
        "broken": bool(prot_broken),
        # break/reclaim_pending: TRUE iff the regime latch flipped ON the decision bar
        # (li) -- i.e. the sim queued the protect SELLs / the Reclaim rebuy on li, to
        # fill at the next open (= the MOO the bridge must submit). The old re-derivation
        # (`not prot_broken and dc < sma*(1-x)`) read the POST-loop latch, which sec G

        # NEVER True on any bar (audit 2026-07-01: protect/reclaim/hedge silently never
        # emitted). Bar-index equality is exact and also covers the no-lots break and the
        # empty-pool reclaim (nothing queued, but the bear hedge must still flip/close).
        "break_pending": bool(p["protectOn"] and prot_break_i == li),
        "reclaim_pending": bool(p["protectOn"] and prot_reclaim_i == li),
        "sell_pct": float(p["protectSellPct"]),
        "protect_pool": round(protect_pool, 2),

        "close": None if is_na(prot_dc) else round(prot_dc, 2),
        "hedge_ticker": _hedge_tkr or None,
        "hedge_px": None if _hpx is None else round(_hpx, 2),
        "hedge_shares": round(hedge_shares, 4),
        "hedge_active": bool(prot_broken),
    }
    live["hedge_mark"] = round(hedge_shares * _hpx, 2) if _hpx is not None else 0.0

    # lotTrail live export: the per-lot trailing-gain harvest (sec E2) is PER-LOT and
    # PATH-DEPENDENT — each lot trails its OWN peak — so unlike the regime engines the
    # bridge cannot read a single fire flag; it must replay the arm/retrace logic over
    # its REAL (start-flat) lots, whose entries differ from the sim's. Export the
    # EFFECTIVE per-bar params (resolved for ATR / z-slope / chop / exo-window, all
    # collapsed to scalars via the SHARED _eff_trail helper -> no drift vs the loop) so
    # the bridge applies them verbatim with no regime re-derivation. cooldown_bars +
    # min_gain_pct + er_block complete the fire predicate; entry_f_min flags the one
    # per-lot input the bridge can't reconstruct (gated off-submit in loop.py). On a
    # 1d cell the fire reads the daily close -> a next-open MOO, daily-confirmed (NOT
    # an intraday resting stop). Read-only; zero sim behavior change. See bridge/trail.py.
    _lt_in_exo = (int(p["lotTrailExoWindowDays"]) > 0 and
                  (day_count - exo_fire_day) < int(p["lotTrailExoWindowDays"]))
    _lt_arm, _lt_retr, _lt_frac = _eff_trail(
        p, trail_atr_held[li], z_held[li], er_held[li], _lt_in_exo)
    _lt_er_block = (float(p["lotTrailErMax"]) > 0 and not is_na(er_held[li])
                    and er_held[li] > float(p["lotTrailErMax"]))
    # NB: every *_pct field is in PERCENT (e.g. 5.0 == 5%); bridge/trail.py divides
    # each by 100 before comparing to the fractional lot gain / retrace.
    live["lot_trail"] = {
        "on": bool(p["lotTrailOn"]),
        "arm_pct": round(_lt_arm * 100, 6),
        "retrace_pct": round(_lt_retr * 100, 6),
        "sell_frac_pct": round(_lt_frac * 100, 6),

        "entry_f_min": float(p["lotTrailEntryFMin"]),
    }

    z_er_gate = (float(p["zSellErMax"]) > 0 and not is_na(er_held[li])
                 and er_held[li] > float(p["zSellErMax"]))
    z_ladder = []
    if bool(p["zSellOn"]):
        for tn in (1, 2, 3):
            if not p[f"zt{tn}En"]:
                continue

                "tier": tn,
                "armed": bool(zt_armed[tn]),
                "z_thresh": th,
                "dose_pct": float(p[f"zt{tn}Pct"]),
                "er_gated": gated,
                "cooldown_ok": bool(cd_ok),                # informational
                # fire_pending: snapshot-gated (the sim tags tier tn's sell "Trim z{th:g}").
                # Fires IFF the sim queued this tier on li -> arm/z>=th/cooldown/ER-gate/lots
                # all self-respected; no current-bar z_now re-derivation (that led the sim by
                # a bar + re-armed spuriously). See fired_sell_tags above.
                # zt_last[tn] == day_tomorrow-1 (= li's day_count) pins the fire to THIS
                # tier: two tiers can share one z threshold (bear-qqq zt1Z == zt3Z == 1.4)
                # and then share the ORDER TAG, so the tag test alone marks a tier whose
                # own cooldown kept it out (emitting 36%+8% doses when the sim sold only 8%).
                "fire_pending": (f"Trim z{th:g}" in fired_sell_tags
                                 and zt_last[tn] == day_tomorrow - 1),

    live["z_ladder"] = z_ladder

    if blow_held is not None:
        vb_cd_ok = (day_tomorrow - vb_last) >= int(p["volBlowCooldownDays"])

            "on": True,
            # snapshot-gated (sim tag "VolBlowoff"): fires IFF the sim queued it on li ->
            # cooldown + br.lots self-respected, no current-bar blow_held[li] re-derivation.
            "fire_pending": "VolBlowoff" in fired_sell_tags,

            "cooldown_ok": bool(vb_cd_ok),                 # informational
        }
    else:
        live["volblow"] = {"on": False, "fire_pending": False,
                           "dose_pct": float(p["volBlowPct"]), "cooldown_ok": True}

    return {
        "metrics": metrics,
        "signal_counts": signal_counts,
        "sell_counts": sell_counts,
        "lots": open_lots,
        "attribution": {k: {kk: round(vv, 2) if isinstance(vv, float) else vv for kk, vv in v.items()}
                        for k, v in sorted(attribution.items())},
        "trades": br.trades,
        "fills": br.fills,
        "series": {
            "ts": chart.ts[w0:w1 + 1],
            "open": chart.open[w0:w1 + 1], "high": chart.high[w0:w1 + 1],
            "low": chart.low[w0:w1 + 1], "close": chart.close[w0:w1 + 1],
            "wvf": wvf[w0:w1 + 1],
            "z": z_held[w0:w1 + 1],
            "equity": eq_close[w0:w1 + 1],
            "bench": bench_eq[w0:w1 + 1],
            "levels": {k: {"h": lvl_series[k]["h"][w0:w1 + 1], "l": lvl_series[k]["l"][w0:w1 + 1]}

        "smc_boxes": smc_created,

    }


# ========== supertrend.py ==========
"""Reaction-fitness MULTI-TIMEFRAME trendline optimizer for the 'super' chart.

Where trendlines.py runs a fixed single-pass heuristic per timeframe and
emit_super.py then *merges* those per-TF lines, this module does the opposite and
stronger thing: it SEARCHES the trendline space directly against an objective and
keeps only lines that beat chance.

The objective (chosen with the user, 2026-06-16, vnext-style — vnext has a backtest
PF to maximize; a viz-only trendline has none, so we define one):

    fitness(line) = Σ over ALL timeframes of independent reactions (touches),
                    timeframe-weighted, for lines that PASS a null-model gate.

Why the null-model gate is load-bearing: "maximize touches" alone is trivially
gamed — widen the band or add pivots and any line collects more contacts. So every
candidate is compared against N_NULL random same-slope lines dropped on random
same-side anchors; only lines whose real contact count beats that random baseline
by >= Z_MIN sigma survive (an apophenia / "is-it-luck" test). What's left is then
ranked by timeframe-weighted independent touches and de-duplicated (NMS).

Pipeline per side (support / resistance):
  1. SEED   — all-pairs lines from each TF's most-prominent pivots (a line is a
              calendar-axis object: anchor unix-ts + log-price slope per second, so
              a line seeded on 1d is scored on 2h/4h/1d alike).
  2. PREFILTER — cheap raw cross-TF contact count; keep the top SEED_KEEP seeds.
  3. REFIT  — least-squares re-fit each seed to its inlier pivots across all TFs
              (a line fit to 6 pivots beats a line drawn through just 2).
  4. GATE   — no-violation gate (closes may not break through) + null-model z-gate.
  5. SELECT — rank by weighted fitness, NMS, keep TOP_OPT.

Viz-only and additive — report.py and the strategy/backtest are untouched. The
runner does NOT replace the super chart's own merged trendlines; it only ADDS the
optimizer's lines that are genuinely NEW (not already represented by a super line)
onto the existing overlays-<sym>-super.js, tagged `supersuper` so the viewer draws
them in their own colour. Run AFTER emit_super:

    py fearlab/supertrend.py            # SPY: add supersuper lines to spy-super
    py fearlab/supertrend.py <dir>      # explicit dashboard dir (tests)

NOTE: this is STAGE 3 of the chart-overlay pipeline — it augments what emit_super
(stage 2) wrote, so emit_super must have run first on the SAME dashboard. To regen
overlays, run fearlab/build_overlays.py, which runs all stages in order.
"""

def _prep_tf(candles):
    """Per-TF arrays the search reuses: log OHLC, touch band, swing pivots."""
    candles = [c for c in candles if c.get("low", 0) > 0 and c.get("high", 0) > 0]
    n = len(candles)
    ts = [c["time"] for c in candles]
    high = [c["high"] for c in candles]
    low = [c["low"] for c in candles]
    lhigh = [math.log(x) for x in high]
    llow = [math.log(x) for x in low]
    lclose = [math.log(c["close"]) for c in candles]
    band = T.BAND_ATR_MULT * T._median([lhigh[i] - llow[i] for i in range(n)])
    return {"n": n, "ts": ts, "lhigh": lhigh, "llow": llow, "lclose": lclose,
            "band": band,
            "piv_hi": T._pivots(high, T.PIVOT_LOOKBACK, True),
            "piv_lo": T._pivots(low, T.PIVOT_LOOKBACK, False)}


def _side(tf, support):
    """(pivots, log-price array) for one side of a TF."""
    return (tf["piv_lo"], tf["llow"]) if support else (tf["piv_hi"], tf["lhigh"])


def _val(t0, b0, m, t):
    return b0 + m * (t - t0)


def _raw_contacts(tfs, support, t0, b0, m):
    """Cheap cross-TF contact count: pivots within band of the line, no independence
    test. Used for the prefilter AND as the null-model statistic (compared like-for-like
    against random lines), so the expensive independence scan stays off the hot path."""
    tot = 0
    for tf in tfs.values():
        piv, lp = _side(tf, support)
        band = tf["band"]
        ts = tf["ts"]
        tot += sum(1 for i in piv if abs(lp[i] - _val(t0, b0, m, ts[i])) <= band)
    return tot


def _refit(tfs, support, t0, b0, m):
    """Least-squares re-fit the line to its inlier pivots (log-price vs seconds-from-t0)
    across all TFs, re-selecting inliers each pass. Returns (b0, m) anchored at t0."""
    for _ in range(REFIT_ITERS):
        sx = sy = sxx = sxy = 0.0
        k = 0
        for tf in tfs.values():
            piv, lp = _side(tf, support)
            band = tf["band"]
            ts = tf["ts"]
            for i in piv:
                x = ts[i] - t0
                if abs(lp[i] - _val(t0, b0, m, ts[i])) <= band:
                    sx += x; sy += lp[i]; sxx += x * x; sxy += x * lp[i]; k += 1
        if k < 3:
            return b0, m
        den = k * sxx - sx * sx
        if den == 0:
            return b0, m
        m = (k * sxy - sx * sy) / den
        b0 = (sy - m * sx) / k          # intercept at x = 0, i.e. at t0
    return b0, m


def _viol_ok(tfs, support, t0, b0, m):
    """A line is respected if the FRACTION of its in-span closes that break through it
    (by more than VIOL_TOL * band on the wrong side) is small. Rate-based, not an
    absolute count, so a clean multi-year line isn't punished for having more bars than
    a short one. Pooled across TFs."""
    v = 0
    span_bars = 0
    for tf in tfs.values():
        ts = tf["ts"]
        lclose = tf["lclose"]
        band = tf["band"] * VIOL_TOL
        for i in range(tf["n"]):
            if ts[i] < t0:
                continue
            span_bars += 1
            d = lclose[i] - _val(t0, b0, m, ts[i])
            if (support and d < -band) or (not support and d > band):
                v += 1
    return v <= max(MAX_VIOL_ABS, MAX_VIOL_FRAC * span_bars)


def _fitness(tfs, support, t0, b0, m):
    """Independent reactions across all TFs. Returns (total_indep, weighted, tf_hits,
    last_touch_ts) — weighted scales each TF's count by TF_WEIGHTS, tf_hits = #TFs that
    confirm the line (>= MIN_TF_TOUCH independent touches), last_touch_ts = most recent
    independent touch anywhere (for the still-active gate)."""
    total = 0
    weighted = 0.0
    tf_hits = 0
    last_ts = -1
    for name, tf in tfs.items():
        piv, lp = _side(tf, support)
        ts = tf["ts"]
        la = lambda i, ts=ts: _val(t0, b0, m, ts[i])
        cnt, last = T._independent_touches(piv, lp, tf["lclose"], la, tf["band"], support)
        total += cnt
        weighted += TF_WEIGHTS.get(name, 1.0) * cnt
        if cnt >= MIN_TF_TOUCH:
            tf_hits += 1
        if last >= 0:
            last_ts = max(last_ts, ts[last])
    return total, weighted, tf_hits, last_ts


def _null_z(tfs, support, m, real_raw, rng):
    """z-score of the line's raw contact count against N_NULL random same-slope lines
    anchored on random same-side pivots. High z = far more contacts than a random line
    of identical slope would collect here = unlikely to be coincidence."""
    anchors = []
    for tf in tfs.values():
        piv, lp = _side(tf, support)
        ts = tf["ts"]
        anchors.extend((ts[i], lp[i]) for i in piv)
    if len(anchors) < 8:
        return 0.0
    samples = []
    for _ in range(N_NULL):
        t0r, b0r = rng.choice(anchors)
        samples.append(_raw_contacts(tfs, support, t0r, b0r, m))
    mean = sum(samples) / len(samples)
    var = sum((s - mean) ** 2 for s in samples) / len(samples)
    sd = math.sqrt(var)
    return (real_raw - mean) / sd if sd > 1e-9 else (99.0 if real_raw > mean else 0.0)


def _optimize_side(tfs, support, rng, min_m, seed_after_ts=None):
    """Full search for one side. Returns kept candidate dicts (calendar-axis lines).
    min_m = slope-magnitude floor: seeds flatter than this are pseudo-levels, skipped.
    seed_after_ts = if set, only seed lines whose EARLIER anchor is on/after this time
    (the recent near-price pass restricts seeding to recent pivots).

    Note: there is deliberately NO contact-based prefilter — ranking seeds by raw
    contacts selects oscillating lines (many contacts AND many violations) and starves
    the clean trendlines (fewer contacts, ~zero violations). Instead every (deduped)
    seed is refit + violation-gated + scored, ranked by fitness, and only the top
    NULL_KEEP pay for the expensive null-model test."""
    # 1. SEED: all-pairs from each TF's most-prominent pivots (diagonal lines only)
    seeds = []
    for tf in tfs.values():
        piv, lp = _side(tf, support)
        ts = tf["ts"]
        ranked = sorted(piv, key=lambda i: T._prominence(lp, i, T.PIVOT_LOOKBACK, not support),
                        reverse=True)[:MAX_SEED_PIVOTS]
        idx = sorted(ranked)
        for ai in range(len(idx)):
            a = idx[ai]
            if seed_after_ts is not None and ts[a] < seed_after_ts:
                continue                  # earlier anchor outside the recent window
            for b in idx[ai + 1:]:
                if ts[b] - ts[a] <= 0:
                    continue
                m = (lp[b] - lp[a]) / (ts[b] - ts[a])
                if abs(m) < min_m:        # near-flat -> a level, not a trendline
                    continue
                seeds.append((ts[a], lp[a], m))
    # 2. DEDUP near-identical seeds (slope bucket + level at the last bar), then refit +
    #    violation-gate + score each survivor.
    t_ref = next(iter(tfs.values()))["ts"][-1]
    seen = set()
    evald = []
    for t0, b0, m in seeds:
        k = (round(m / min_m), round(_val(t0, b0, m, t_ref), 3))
        if k in seen:
            continue
        seen.add(k)
        b0, m = _refit(tfs, support, t0, b0, m)
        if abs(m) < min_m or not _viol_ok(tfs, support, t0, b0, m):
            continue
        total, weighted, tf_hits, last_ts = _fitness(tfs, support, t0, b0, m)
        if total < T.MIN_TOUCHES:
            continue
        evald.append({"t0": t0, "b0": b0, "m": m, "touches": total, "weighted": weighted,
                      "tfs": tf_hits, "last_ts": last_ts})
    # 3. null-model test on the top fitness-ranked clean lines only
    evald.sort(key=lambda c: c["weighted"], reverse=True)
    cands = []
    for c in evald[:NULL_KEEP]:
        raw = _raw_contacts(tfs, support, c["t0"], c["b0"], c["m"])
        z = _null_z(tfs, support, c["m"], raw, rng)
        if z < Z_MIN:
            continue
        c["kind"] = "support" if support else "resistance"
        c["z"] = round(z, 2)
        cands.append(c)
    return cands


def _nms(cands, t_end, band):
    """Keep highest-fitness lines, dropping near-duplicates (similar slope AND similar
    value at the display end). Ranked by `rank` (touch fitness + confluence + span)."""
    kept = []
    for c in sorted(cands, key=lambda c: c["rank"], reverse=True):
        vc = _val(c["t0"], c["b0"], c["m"], t_end)
        dup = False
        for k in kept:
            vk = _val(k["t0"], k["b0"], k["m"], t_end)
            same_slope = abs(c["m"] - k["m"]) <= 0.15 * (abs(k["m"]) + 1e-12) + 1e-12
            if same_slope and abs(vc - vk) <= band and c["kind"] == k["kind"]:
                dup = True
                break
        if not dup:
            kept.append(c)
    return kept


def _finalize(cands, display, t0d, t1d, span_sec, dband, lo, hi, rec_cut, min_m,
              top_n, tag, near_frac, price, exclude):
    """Filter + rank + NMS a candidate pool, then re-anchor survivors to the display
    window as output trendline dicts tagged with `tag`. near_frac (support pass only):
    keep only lines whose current value sits within that fraction BELOW price (near-price).
    exclude = already-emitted lines to dedup against (so passes don't repeat each other)."""
    cands = [c for c in cands if abs(c["m"]) >= min_m]             # refit may have flattened a line
    cands = [c for c in cands if c["tfs"] >= MIN_TF_HITS]          # >= 1 TF with 2+ touches (no scatter)
    cands = [c for c in cands if c["last_ts"] >= rec_cut]          # drop stale lines
    cands = [c for c in cands if lo <= _val(c["t0"], c["b0"], c["m"], t1d) <= hi]  # on-chart now
    if near_frac is not None:                                      # near-price band (supports)
        lo_n, hi_n = math.log(price * (1 - near_frac)), math.log(price) + dband
        cands = [c for c in cands if lo_n <= _val(c["t0"], c["b0"], c["m"], t1d) <= hi_n]
    for c in cands:                                                # rank = fitness + confluence + span
        span = min(1.0, (t1d - c["t0"]) / span_sec)
        c["rank"] = c["weighted"] + 0.5 * c["tfs"] + SPAN_W * span
    dl_close = math.log(display[-1]["close"])
    out = []
    kept_keys = list(exclude)               # dedup new lines against the prior pass + each other
    for c in _nms(cands, t1d, dband):
        if len(out) >= top_n:
            break
        v0 = _val(c["t0"], c["b0"], c["m"], t0d)
        v1 = _val(c["t0"], c["b0"], c["m"], t1d)
        support = c["kind"] == "support"
        line = {"t1": t0d, "p1": round(math.exp(v0), 4),
                "t2": t1d, "p2": round(math.exp(v1), 4),
                "kind": c["kind"], "touches": c["touches"], "tfs": c["tfs"],
                "live": (dl_close >= v1 - dband) if support else (dl_close <= v1 + dband),
                "z": c["z"], tag: True}
        if any(_same_line(line, e, dband) for e in kept_keys):     # dedup vs prior pass + own kept
            continue
        out.append(line)
        kept_keys.append(line)
    return out


def optimize(tf_candles, display):
    """tf_candles = {tf_name: candles}; display = the candle list to draw on. Returns
    trendline dicts (t1,p1,t2,p2,kind,touches,tfs,live,z) from TWO passes:
      * main  — full history, both sides, tagged `supersuper` (drawn pink)
      * recent — supports anchored within RECENT_YEARS and near price, tagged `recent`
                 (drawn white); deduped against the main pass."""
    tfs = {name: _prep_tf(c) for name, c in tf_candles.items() if c}
    if not tfs or not display:
        return []
    t0d, t1d = display[0]["time"], display[-1]["time"]
    span_sec = max(1, t1d - t0d)
    dband = T.BAND_ATR_MULT * T._median(
        [math.log(c["high"]) - math.log(c["low"]) for c in display])
    lo = math.log(min(c["low"] for c in display)) - dband
    hi = math.log(max(c["high"] for c in display)) + dband
    rec_cut = t1d - T.RECENCY_SEC          # still-active: last touch within RECENCY_YEARS
    min_m = MIN_SLOPE_BANDS * dband / span_sec   # slope floor: >= this many bands over the window
    price = display[-1]["close"]

    # each pass gets its OWN seeded rng so the recent pass's null-model z is independent of how
    # many draws the main pass consumed (and reproducible run-to-run).
    main_cands = (_optimize_side(tfs, True, random.Random(NULL_SEED), min_m)
                  + _optimize_side(tfs, False, random.Random(NULL_SEED + 1), min_m))
    main = _finalize(main_cands, display, t0d, t1d, span_sec, dband, lo, hi, rec_cut, min_m,
                     TOP_OPT, "supersuper", None, price, exclude=[])

    recent_cut = t1d - int(RECENT_YEARS * 365.25 * 86400)
    rec_cands = _optimize_side(tfs, True, random.Random(NULL_SEED + 2), min_m, seed_after_ts=recent_cut)
    recent = _finalize(rec_cands, display, t0d, t1d, span_sec, dband, lo, hi, rec_cut, min_m,
                       TOP_RECENT_SUP, "recent", NEAR_FRAC, price, exclude=main)

    return main + recent


def _prominent_side(tfs, support, t0d, t1d, dband, span_sec, min_m, rec_cut):
    """Lines through the FEW most-PROMINENT same-side swing pivots in the display
    window. Clean (violation-gated) and multi-TF-confirmed, but ranked by
    anchor-prominence × span — NOT by touch count — and with NO null-model z-gate.
    This is the line a human draws through the major swing lows/highs (the channel
    edge); the count-based optimizer rejects it because a clean steep line hits about
    as many pivots as a random same-slope line (z≈0)."""
    pool = []   # (ts, log-price, prominence, band) for the most-prominent pivots, pooled across TFs
    for tf in tfs.values():
        piv, lp = _side(tf, support)
        ts = tf["ts"]
        recent = [i for i in piv if ts[i] >= t0d]
        recent.sort(key=lambda i: T._prominence(lp, i, T.PIVOT_LOOKBACK, not support), reverse=True)
        for i in recent[:PROMINENT_TOP_PIVOTS]:
            pool.append((ts[i], lp[i], T._prominence(lp, i, T.PIVOT_LOOKBACK, not support), tf["band"]))
    pool.sort(key=lambda p: p[2], reverse=True)
    pool = pool[:PROMINENT_TOP_PIVOTS]
    if len(pool) < 2:
        return []
    out = []
    seen = set()
    for ai in range(len(pool)):
        ta, ya, _, _ = pool[ai]
        for bi in range(ai + 1, len(pool)):
            tb, yb, _, _ = pool[bi]
            if tb == ta:
                continue
            m = (yb - ya) / (tb - ta)
            if abs(m) < min_m:                                   # near-flat -> a level, not a line
                continue
            b0, m = _refit(tfs, support, ta, ya, m)              # tighten onto inlier pivots
            if abs(m) < min_m or not _viol_ok(tfs, support, ta, b0, m):
                continue
            anchors = prom_sum = 0                                # how many BIG pivots it actually connects
            for tp, yp, pr, bnd in pool:
                if abs(yp - _val(ta, b0, m, tp)) <= bnd:
                    anchors += 1
                    prom_sum += pr
            if anchors < PROMINENT_MIN_ANCHORS:
                continue
            total, weighted, tf_hits, last_ts = _fitness(tfs, support, ta, b0, m)
            if tf_hits < MIN_TF_HITS or last_ts < rec_cut:
                continue
            key = (round(m / min_m), round(_val(ta, b0, m, t1d), 3))   # dedup near-identical seeds
            if key in seen:
                continue
            seen.add(key)
            span = min(1.0, (t1d - ta) / span_sec)
            out.append({"t0": ta, "b0": b0, "m": m, "z": 0.0,
                        "kind": "support" if support else "resistance",
                        "touches": total, "tfs": tf_hits, "last_ts": last_ts,
                        "anchors": anchors, "rank": prom_sum * (1.0 + span)})
    return out


def optimize_prominent(tf_candles, display):
    """Prominent-swings pass — see _prominent_side. Returns trendline dicts tagged
    `prominent` (drawn teal), up to TOP_PROMINENT per side, NMS-deduped."""
    tfs = {name: _prep_tf(c) for name, c in tf_candles.items() if c}
    if not tfs or not display:
        return []
    t0d, t1d = display[0]["time"], display[-1]["time"]
    span_sec = max(1, t1d - t0d)
    dband = T.BAND_ATR_MULT * T._median(
        [math.log(c["high"]) - math.log(c["low"]) for c in display])
    lo = math.log(min(c["low"] for c in display)) - dband
    hi = math.log(max(c["high"] for c in display)) + dband
    rec_cut = t1d - T.RECENCY_SEC
    min_m = MIN_SLOPE_BANDS * dband / span_sec
    dl_close = math.log(display[-1]["close"])

    cands = (_prominent_side(tfs, True, t0d, t1d, dband, span_sec, min_m, rec_cut)
             + _prominent_side(tfs, False, t0d, t1d, dband, span_sec, min_m, rec_cut))
    yr = 365.25 * 86400
    price = display[-1]["close"]
    near_lo, near_hi = math.log(price * (1 - PROMINENT_MAX_DIST)), math.log(price * (1 + PROMINENT_MAX_DIST))
    cands = [c for c in cands if lo <= _val(c["t0"], c["b0"], c["m"], t1d) <= hi
             and abs(math.exp(c["m"] * yr) - 1) * 100 >= PROMINENT_MIN_PCTYR   # drop near-flat (level-like)
             and near_lo <= _val(c["t0"], c["b0"], c["m"], t1d) <= near_hi]    # drop far-from-price (deep/ancient)
    out = []
    per_side = {"support": 0, "resistance": 0}
    for c in _nms(cands, t1d, dband):                            # ranked by prominence×span
        if per_side[c["kind"]] >= TOP_PROMINENT:
            continue
        v0 = _val(c["t0"], c["b0"], c["m"], t0d)
        v1 = _val(c["t0"], c["b0"], c["m"], t1d)
        support = c["kind"] == "support"
        line = {"t1": t0d, "p1": round(math.exp(v0), 4), "t2": t1d, "p2": round(math.exp(v1), 4),
                "kind": c["kind"], "touches": c["touches"], "tfs": c["tfs"],
                "live": (dl_close >= v1 - dband) if support else (dl_close <= v1 + dband),
                "z": 0.0, "anchors": c["anchors"], "prominent": True}
        if any(_same_line(line, o, dband) for o in out):
            continue
        out.append(line)
        per_side[c["kind"]] += 1
    return out


# ---- runner: ADD the optimizer's new lines onto the existing <sym>-super overlay ----
# The optimizer does NOT replace the super's own merged trendlines (user, 2026-06-16);
# it only ADDS lines that are genuinely new (not already represented by a super line)
# and relevant (they already cleared the fitness + null-model gates). Added lines are
# tagged `supersuper` so the viewer draws them in their own colour.

def _read_overlay(path):
    with open(path, encoding="utf-8") as f:
        s = f.read()
    i = s.rindex("]=") + 2
    return json.loads(s[i: s.rfind("}") + 1])


def _same_line(a, b, band):
    """Two super-window lines coincide when they share kind and agree (within band, log
    price) at BOTH display endpoints — they're re-anchored to the same [t0d,t1d], so
    matching p1 and p2 means the same line."""
    return (a["kind"] == b["kind"]
            and abs(math.log(a["p1"]) - math.log(b["p1"])) <= band
            and abs(math.log(a["p2"]) - math.log(b["p2"])) <= band)


def _ln_at(t, at):
    """Log-price of an overlay trendline (endpoints t1/p1, t2/p2) at calendar time `at`."""
    a, b = math.log(t["p1"]), math.log(t["p2"])
    return a if t["t2"] == t["t1"] else a + (b - a) * (at - t["t1"]) / (t["t2"] - t["t1"])


def _layer_of(t):
    return ("recent" if t.get("recent") else "supersuper" if t.get("supersuper")
            else "prominent" if t.get("prominent") else "base")


def _confluence_zones(trendlines, display, band):
    """Where >= CONFLUENCE_MIN_LINES distinct trendlines from >= CONFLUENCE_MIN_LAYERS layers
    stack within `band` at the live edge (last candle). Returns role-tagged horizontal zones
    for the viewer to draw. Drawing aid only — no strength score, no sizing semantics.

    Dedup: lines that coincide at BOTH window ends (same slope+value) collapse to one, so a
    zone's count reflects DISTINCT geometric lines, not the same line found by several passes.
    Clustering: windows of TOTAL span <= band (hard cap) — no single-linkage chaining."""
    if not trendlines or not display:
        return []
    t0, t1 = display[0]["time"], display[-1]["time"]
    price = display[-1]["close"]
    dtol = 0.3 * band
    kept = []
    for t in sorted(trendlines, key=lambda x: -(x.get("touches", 0) or 0)):
        a0, a1 = _ln_at(t, t0), _ln_at(t, t1)
        if any(abs(a0 - _ln_at(k, t0)) < dtol and abs(a1 - _ln_at(k, t1)) < dtol for k in kept):
            continue
        kept.append(t)
    cap = band * CONFLUENCE_BAND_MULT
    vals = sorted(((_ln_at(t, t1), t) for t in kept), key=lambda x: x[0])
    zones, i = [], 0
    while i < len(vals):
        j = i
        while j + 1 < len(vals) and vals[j + 1][0] - vals[i][0] <= cap:
            j += 1
        g = vals[i:j + 1]
        layers = {_layer_of(t) for _, t in g}
        if len(g) >= CONFLUENCE_MIN_LINES and len(layers) >= CONFLUENCE_MIN_LAYERS:
            cp = math.exp(sum(x for x, _ in g) / len(g))
            zones.append({"price": round(cp, 4),
                          "lo": round(math.exp(g[0][0]), 4), "hi": round(math.exp(g[-1][0]), 4),
                          "n": len(g), "layers": len(layers),
                          "role": "support" if cp <= price else "resistance"})
        i = j + 1
    return zones


def augment_super(dash_dir, sym, display_tf, tfs):
    """Inject the optimizer's lines into overlays-<sym>-super.js, deduped against the
    base super lines. optimize() already tags each line `supersuper` (pink, main pass)
    or `recent` (white, recent near-price support pass). Runs AFTER emit_super (which
    writes the base overlay). Idempotent: strips any prior tagged lines before re-adding,
    so re-running without emit_super won't double up."""
    super_path = os.path.join(dash_dir, f"overlays-{sym}-super.js")
    if not os.path.exists(super_path):
        print(f"  {sym}: no overlays-{sym}-super.js (run emit_super first) — skip")
        return False
    tf_candles = {tf: _find_candles(dash_dir, sym, tf) for tf in tfs}
    display = _find_candles(dash_dir, sym, display_tf)
    have = [tf for tf, c in tf_candles.items() if c]
    if not display or not have:
        print(f"  {sym}: missing candles — skip")
        return False

    ov = _read_overlay(super_path)
    base = [t for t in ov.get("trendlines", [])
            if not t.get("supersuper") and not t.get("recent") and not t.get("prominent")]  # idempotent
    opt = optimize(tf_candles, display)
    dband = T.BAND_ATR_MULT * T._median(
        [math.log(c["high"]) - math.log(c["low"]) for c in display])

    new = [o for o in opt if not any(_same_line(o, b, dband) for b in base)]
    # prominent-swings pass: an INDEPENDENT human-eye layer (own toggle), deduped only
    # against the base merged lines — left to overlap the count-optimizer lines so each
    # layer is its own lens the viewer can toggle on/off.
    prom = optimize_prominent(tf_candles, display) if sym in PROMINENT_SYMS else []
    prom = [o for o in prom if not any(_same_line(o, b, dband) for b in base)]
    ov["trendlines"] = base + new + prom
    ov["confluences"] = (_confluence_zones(ov["trendlines"], display, dband)
                         if sym in CONFLUENCE_SYMS else [])

    key = f"{sym}-super"
    with open(super_path, "w", encoding="utf-8") as f:
        f.write("window.FEARLAB_OVERLAYS=window.FEARLAB_OVERLAYS||{};"
                f"window.FEARLAB_OVERLAYS[{json.dumps(key)}]={json.dumps(ov, separators=(',', ':'))};")
    n_pink = sum(1 for o in new if o.get("supersuper"))
    n_white = sum(1 for o in new if o.get("recent"))
    print(f"  {key}: +{n_pink} supersuper(pink) +{n_white} recent(white) +{len(prom)} prominent(teal) "
          f"+{len(ov['confluences'])} confluence on {len(base)} base "
          f"(from {'+'.join(have)}; {len(opt) - len(new)}/{len(opt)} already represented)")
    for z in ov["confluences"]:
        print(f"      confl {z['role'][:3]} @{z['price']:.2f} x{z['n']} lines/{z['layers']}layers")
    for o in new:
        tag = "white-rec" if o.get("recent") else "pink-ss  "
        print(f"      {tag} {o['kind'][:3]} p2={o['p2']:.2f} x{o['touches']} {o['tfs']}TF z={o['z']}")
    for o in prom:
        print(f"      teal-prom {o['kind'][:3]} p2={o['p2']:.2f} x{o['touches']} {o['tfs']}TF "
              f"anchors={o['anchors']}")
    return True


def main(dash_dir=DASH):
    n = 0
    for sym, disp, tfs in OPTS:
        if augment_super(dash_dir, sym, disp, tfs):
            n += 1
    print(f"\naugmented {n} super overlay(s) with optimizer lines -> {dash_dir}")
    return n


if __name__ == "__main__":
    main(sys.argv[1] if len(sys.argv) > 1 else DASH)


# ========== broker.py ==========
"""FearLab broker: per-lot, finite-cash TV emulator.

Deliberate divergences from the v48 Pine source (documented in README):
- Per-lot accounting: every fill is its own lot; sells close a % of ONE lot
  (Pine's strategy.close(id) hits the whole entry-ID aggregate).
- No silent leverage: buy cost is capped at available cash (TV with margin 0
  never rejects orders). Clipped/skipped orders are counted.
- External contributions: contribUsd lands in cash on the first bar of each
  month, like the user wiring $1k/mo into the account.

Fill model (TV semantics):
- market orders queued at signal close, filled at next bar open +- slippage.
- limit buys: filled at min(open, limit) when open <= limit or low <= limit.
- stops (per engine): filled at stop, or at open when the bar gaps through.
"""
from __future__ import annotations

import math


class Lot:
    __slots__ = ("serial", "engine", "cls", "qty", "entry_price", "entry_ts", "entry_idx")

    def __init__(self, serial, engine, cls, qty, entry_price, entry_ts, entry_idx):
        self.serial = serial
        self.engine = engine          # "Fear T1", "L3 Sniper", ...
        self.cls = cls                # "grid" | "sniper"
        self.qty = qty
        self.entry_price = entry_price
        self.entry_ts = entry_ts
        self.entry_idx = entry_idx


class Broker:
    def __init__(self, initial_capital: float, commission_pct: float = 0.0,
                 slippage: float = 0.0, allow_leverage: bool = False,
                 reserve_pct: float = 0.0):
        self.cash = initial_capital
        self.initial = initial_capital
        self.commission_pct = commission_pct
        self.slippage = slippage          # absolute $ per share
        self.allow_leverage = allow_leverage
        # dry powder: reserve_pct% of equity is untouchable except by orders
        # flagged unlock_reserve (extreme-fear buys)
        self.reserve_pct = reserve_pct
        self.lots: list[Lot] = []
        self._serial = 0
        # pending market orders: list of dicts queued at signal close
        self.pending_buys: list[dict] = []
        self.pending_sells: list[dict] = []   # {"lot": Lot, "frac": 0..1, "tag": str}
        self.working_limits: dict[str, dict] = {}   # engine -> {"limit","qty","tag"}
        self.engine_stops: dict[str, float] = {}    # engine -> stop price
        # ledgers
        self.trades: list[dict] = []      # closed (partial) lots
        self.fills: list[dict] = []       # every buy fill (for markers)
        self.flows: list[tuple[int, float]] = []  # (ts, +in/-out) external money
        self.buy_orders = 0
        self.buys_clipped = 0
        self.buys_skipped = 0
        # bear lock (opt-in; set per-bar by the strategy). While True, queue_buy /
        # set_limit drop any order whose tag is not in buy_block_exempt -> "go to
        # cash and stay there" while the protect regime is broken. Default off = inert.
        self.buy_block = False
        self.buy_block_exempt = ("Reclaim",)

    # ---- account ----
    def contribute(self, ts: int, usd: float):
        self.cash += usd
        self.flows.append((ts, usd))

    def position_qty(self) -> float:
        return sum(l.qty for l in self.lots)

    def equity(self, price: float) -> float:
        return self.cash + self.position_qty() * price

    # ---- order placement (at signal close) ----
    def queue_buy(self, engine: str, cls: str, qty: float, tag: str, stop: float | None = None,
                  unlock_reserve: bool = False):
        if self.buy_block and tag not in self.buy_block_exempt:
            self.buys_skipped += 1
            return
        self.pending_buys.append({"engine": engine, "cls": cls, "qty": qty, "tag": tag,
                                  "stop": stop, "unlock": unlock_reserve})

    def queue_sell_lot(self, lot: Lot, frac: float, tag: str):
        self.pending_sells.append({"lot": lot, "frac": frac, "tag": tag})

    def set_limit(self, engine: str, cls: str, limit: float, qty: float, tag: str, stop: float | None = None):
        if self.buy_block and tag not in self.buy_block_exempt:
            return
        self.working_limits[engine] = {"cls": cls, "limit": limit, "qty": qty, "tag": tag, "stop": stop}

    def cancel_limit(self, engine: str):
        self.working_limits.pop(engine, None)

    # ---- fills ----
    def _buy_fill(self, engine, cls, qty, price, ts, idx, tag, stop, unlock=False):
        cost_per_share = price * (1 + self.commission_pct / 100)
        want = qty
        if not self.allow_leverage:
            avail = self.cash
            if self.reserve_pct > 0 and not unlock:
                avail = max(0.0, self.cash - self.equity(price) * self.reserve_pct / 100)
            affordable = avail / cost_per_share if cost_per_share > 0 else 0.0
            if affordable <= 0 or want * cost_per_share > avail:
                if affordable * price < 1.0:   # < $1 -> skip entirely
                    self.buys_skipped += 1
                    return
                want = affordable
                self.buys_clipped += 1
        self.cash -= want * cost_per_share
        self._serial += 1
        lot = Lot(self._serial, engine, cls, want, price, ts, idx)
        self.lots.append(lot)
        self.fills.append({"ts": ts, "idx": idx, "engine": engine, "qty": want,
                           "price": price, "usd": want * price, "tag": tag})
        if stop is not None:
            self.engine_stops[engine] = stop
        self.buy_orders += 1

    def _sell_fill(self, lot: Lot, frac: float, price: float, ts, idx, tag):
        frac = min(max(frac, 0.0), 1.0)
        q = lot.qty * frac
        if q <= 0:
            return
        proceeds = q * price * (1 - self.commission_pct / 100)
        self.cash += proceeds
        pnl = (price - lot.entry_price) * q
        self.trades.append({"ts": ts, "idx": idx, "engine": lot.engine, "qty": q,
                            "entry": lot.entry_price, "exit": price, "pnl": pnl,
                            "pnl_pct": (price / lot.entry_price - 1) * 100, "tag": tag,
                            "entry_ts": lot.entry_ts})
        lot.qty -= q
        if lot.qty * price < 0.01:   # dust
            self.lots.remove(lot)

    def on_bar_open(self, idx: int, ts: int, open_: float):
        """Fill queued market orders at this bar's open. Sells first (frees cash)."""
        for s in self.pending_sells:
            if s["lot"] in self.lots:
                self._sell_fill(s["lot"], s["frac"], open_ - self.slippage, ts, idx, s["tag"])
        self.pending_sells.clear()
        for b in self.pending_buys:
            self._buy_fill(b["engine"], b["cls"], b["qty"], open_ + self.slippage, ts, idx,
                           b["tag"], b["stop"], b.get("unlock", False))
        self.pending_buys.clear()

    def on_bar_intrabar(self, idx: int, ts: int, open_: float, high: float, low: float):
        """Stops (sell) then resting limit buys, intrabar."""
        # engine stops: close every lot of that engine
        for engine, stop in list(self.engine_stops.items()):
            elots = [l for l in self.lots if l.engine == engine]
            if not elots:
                self.engine_stops.pop(engine, None)
                continue
            if open_ <= stop:
                px = open_
            elif low <= stop:
                px = stop
            else:
                continue
            for l in elots:
                self._sell_fill(l, 1.0, px, ts, idx, "SL Hard")
            self.engine_stops.pop(engine, None)
        # limit buys
        for engine, w in list(self.working_limits.items()):
            limit = w["limit"]
            if open_ <= limit:
                px = open_
            elif low <= limit:
                px = limit
            else:
                continue
            self._buy_fill(engine, w["cls"], w["qty"], px, ts, idx, w["tag"], w["stop"])
            self.working_limits.pop(engine, None)

    # ---- metrics helpers ----
    def realized_pnl(self) -> float:
        return sum(t["pnl"] for t in self.trades)


def xirr(flows: list[tuple[int, float]], final_ts: int, final_value: float) -> float:
    """Annualized money-weighted return. flows: (ts, amount_in>0). Bisection."""
    if not flows:
        return float("nan")
    t0 = flows[0][0]
    items = [(ts, -amt) for ts, amt in flows]          # money in = negative CF
    items.append((final_ts, final_value))

    def npv(rate: float) -> float:
        s = 0.0
        for ts, cf in items:
            yrs = (ts - t0) / (365.25 * 86400)
            s += cf / ((1 + rate) ** yrs)
        return s

    lo, hi = -0.95, 10.0
    if npv(lo) * npv(hi) > 0:
        return float("nan")
    for _ in range(200):
        mid = (lo + hi) / 2
        if npv(lo) * npv(mid) <= 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2


def max_drawdown(equity_high: list[float], equity_low: list[float]) -> float:
    """Intrabar-approximated max DD %: trough on low-marked equity vs running peak
    of high-marked equity (peak as of that time, not final peak)."""
    peak = -math.inf
    mdd = 0.0
    for i in range(len(equity_high)):
        if equity_high[i] > peak:
            peak = equity_high[i]
        if peak > 0:
            dd = (peak - equity_low[i]) / peak * 100
            if dd > mdd:
                mdd = dd
    return mdd


# ========== data.py ==========
"""FearLab data layer: Yahoo chart API fetch + cache, intraday resample.

Chart TFs "2h"/"4h" are built from Yahoo 60m bars (Yahoo has no native 2h/4h):
session-aligned resample per trading day -> 2h: 9:30 / 11:30 / 13:30 / 15:30,
4h: 9:30 / 13:30 buckets, matching TradingView's RTH bars. 60m history is
capped by Yahoo at ~730 days (the local cache reaches further back).

Daily ("1d") fetches full history (needed for W/M/3M engines' warmup).
Cache: fearlab/data/<symbol>_<interval>.csv  (epoch,open,high,low,close,volume)
"""
from __future__ import annotations

import csv
import json
import os
import subprocess
import time
import urllib.parse

DATA_DIR = os.path.join(os.path.dirname(__file__), "data")
UA = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) fearlab/1.0"


class Bars:
    __slots__ = ("ts", "open", "high", "low", "close", "volume")

    def __init__(self):
        self.ts: list[int] = []
        self.open: list[float] = []
        self.high: list[float] = []
        self.low: list[float] = []
        self.close: list[float] = []
        self.volume: list[float] = []

    def __len__(self):
        return len(self.ts)

    def append(self, t, o, h, l, c, v):
        self.ts.append(t)
        self.open.append(o)
        self.high.append(h)
        self.low.append(l)
        self.close.append(c)
        self.volume.append(v)


def _yahoo_fetch(symbol: str, interval: str) -> Bars:
    if interval == "1d":
        span = "period1=0&period2=9999999999"
    else:
        # intraday: Yahoo caps 60m at 730 days; ask for the max window
        span = "range=730d"
    url = (
        f"https://query1.finance.yahoo.com/v8/finance/chart/{urllib.parse.quote(symbol)}"
        f"?interval={interval}&{span}"
    )
    # TinyWall blocks Python sockets (WinError 10013); curl.exe is allowed.
    proc = subprocess.run(
        ["curl.exe", "-s", "--max-time", "60", "-A", UA, url],
        capture_output=True, text=True, check=True,
    )
    payload = json.loads(proc.stdout)
    result = payload["chart"]["result"][0]
    ts = result["timestamp"]
    q = result["indicators"]["quote"][0]
    bars = Bars()
    for i, t in enumerate(ts):
        o, h, l, c = q["open"][i], q["high"][i], q["low"][i], q["close"][i]
        v = q["volume"][i]
        if o is None or h is None or l is None or c is None:
            continue
        bars.append(int(t), float(o), float(h), float(l), float(c), float(v or 0))
    return bars


def _cache_path(symbol: str, interval: str) -> str:
    safe = symbol.replace("^", "_").replace("/", "_")
    return os.path.join(DATA_DIR, f"{safe}_{interval}.csv")


def _seed_path(symbol: str, interval: str) -> str:
    """A frozen historical underlay fetched ONCE from a deeper source than Yahoo
    (e.g. Databento ohlcv-1m -> 60m, back to 2018). Kept SEPARATE from the live
    cache so the wholesale overwrite in _fetch_cached can't clobber it. Same
    schema as the cache (ts,open,high,low,close,volume; ts = epoch seconds, UTC;
    sorted asc). See fearlab/seed_intraday.py for the builder."""
    safe = symbol.replace("^", "_").replace("/", "_")
    return os.path.join(DATA_DIR, f"{safe}_{interval}_seed.csv")


def _save(path: str, bars: Bars):
    os.makedirs(DATA_DIR, exist_ok=True)
    with open(path, "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["ts", "open", "high", "low", "close", "volume"])
        for i in range(len(bars)):
            w.writerow([bars.ts[i], bars.open[i], bars.high[i], bars.low[i], bars.close[i], bars.volume[i]])


def _load(path: str) -> Bars:
    bars = Bars()
    with open(path, newline="") as f:
        r = csv.reader(f)
        next(r)
        for row in r:
            bars.append(int(row[0]), float(row[1]), float(row[2]), float(row[3]), float(row[4]), float(row[5]))
    return bars


def _merge_save(path: str, seed_path: str, fresh: Bars) -> Bars:
    """Union a frozen deep-history seed with freshly-fetched bars, write to `path`.

    FRESH WINS on overlapping ts (recent Yahoo bars are the authoritative,
    consolidated source; the seed only fills the pre-Yahoo past). Output is sorted
    ascending and deduped by ts -- the invariants resample()/simulate() rely on.
    This is what lets a one-time deep intraday seed survive the wholesale overwrite
    that plain _save does on every refresh."""
    merged: dict[int, tuple] = {}
    if os.path.exists(seed_path):
        seed = _load(seed_path)
        for i in range(len(seed)):
            merged[seed.ts[i]] = (seed.open[i], seed.high[i], seed.low[i],
                                  seed.close[i], seed.volume[i])
    for i in range(len(fresh)):                     # fresh overlays seed on collisions
        merged[fresh.ts[i]] = (fresh.open[i], fresh.high[i], fresh.low[i],
                               fresh.close[i], fresh.volume[i])
    out = Bars()
    for t in sorted(merged):
        o, h, l, c, v = merged[t]
        out.append(t, o, h, l, c, v)
    _save(path, out)
    return out


def _fetch_cached(symbol: str, interval: str, refresh: bool, max_age_hours: float = 24.0) -> Bars:
    path = _cache_path(symbol, interval)
    if not refresh and os.path.exists(path):
        age_h = (time.time() - os.path.getmtime(path)) / 3600
        if age_h <= max_age_hours:
            return _load(path)
    try:
        bars = _yahoo_fetch(symbol, interval)
    except Exception:
        if os.path.exists(path):  # offline -> stale cache beats nothing
            return _load(path)
        raise
    # If a frozen deep-history seed exists (e.g. Databento-built SPY_60m_seed.csv),
    # union it under the fresh Yahoo bars so the deep past persists across refreshes.
    seed = _seed_path(symbol, interval)
    if os.path.exists(seed):
        return _merge_save(path, seed, bars)
    _save(path, bars)
    return bars


def _utc_date(ts: int) -> str:
    # SPY RTH session (13:30-21:00 UTC) never crosses UTC midnight,
    # so grouping by UTC date == grouping by trading day.
    return time.strftime("%Y-%m-%d", time.gmtime(ts))


def resample(bars: Bars, bucket_seconds: int) -> Bars:
    """Session-aligned resample: bucket index = (ts - day_open_ts) // bucket_seconds.

    Handles half days and missing bars; bucket boundaries anchor to each day's
    first bar (9:30 ET), matching TV's intraday bar alignment.
    """
    out = Bars()
    cur_day = None
    day_open_ts = 0
    cur_bucket = -1
    o = h = l = c = v = 0.0
    t0 = 0

    def flush():
        if cur_bucket >= 0:
            out.append(t0, o, h, l, c, v)

    for i in range(len(bars)):
        d = _utc_date(bars.ts[i])
        if d != cur_day:
            flush()
            cur_day = d
            day_open_ts = bars.ts[i]
            cur_bucket = -1
        b = (bars.ts[i] - day_open_ts) // bucket_seconds
        if b != cur_bucket:
            flush()
            cur_bucket = b
            t0 = bars.ts[i]
            o, h, l, c, v = bars.open[i], bars.high[i], bars.low[i], bars.close[i], bars.volume[i]
        else:
            h = max(h, bars.high[i])
            l = min(l, bars.low[i])
            c = bars.close[i]
            v += bars.volume[i]
    flush()
    return out


def get_dividends(symbol: str, refresh: bool = False) -> dict:
    """Ex-date epoch -> per-share dividend (Yahoo events=div), cached as
    fearlab/data/<symbol>_div.csv. Used by the divOn honesty diagnostic."""
    path = _cache_path(symbol, "div")
    if not refresh and os.path.exists(path):
        age_h = (time.time() - os.path.getmtime(path)) / 3600
        if age_h <= 24.0:
            out = {}
            with open(path, newline="") as f:
                r = csv.reader(f)
                next(r)
                for row in r:
                    out[int(row[0])] = float(row[1])
            return out
    url = (
        f"https://query1.finance.yahoo.com/v8/finance/chart/{urllib.parse.quote(symbol)}"
        f"?interval=1d&period1=0&period2=9999999999&events=div"
    )
    proc = subprocess.run(
        ["curl.exe", "-s", "--max-time", "60", "-A", UA, url],
        capture_output=True, text=True, check=True,
    )
    payload = json.loads(proc.stdout)
    evs = payload["chart"]["result"][0].get("events", {}).get("dividends", {}) or {}
    out = {int(v["date"]): float(v["amount"]) for v in evs.values()}
    os.makedirs(DATA_DIR, exist_ok=True)
    with open(path, "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["ts", "amount"])
        for t in sorted(out):
            w.writerow([t, out[t]])
    return out


def get_chart_bars(symbol: str, tf: str, refresh: bool = False) -> Bars:
    """tf: '2h'/'4h' (60m resampled), '1h', '1d'."""
    if tf == "1d":
        return _fetch_cached(symbol, "1d", refresh)
    if tf == "1h":
        return _fetch_cached(symbol, "60m", refresh)
    if tf == "2h":
        return resample(_fetch_cached(symbol, "60m", refresh), 7200)
    if tf == "4h":
        return resample(_fetch_cached(symbol, "60m", refresh), 14400)
    raise SystemExit(f"unsupported tf: {tf} (use 2h, 4h, 1h, 1d)")


def get_daily(symbol: str, refresh: bool = False) -> Bars:
    return _fetch_cached(symbol, "1d", refresh)


if __name__ == "__main__":

def _wma(src: list[float], length: int) -> list[float]:
    out = [NAN] * len(src)
    denom = length * (length + 1) / 2
    for i in range(len(src)):
        if i < length - 1:
            continue
        s = 0.0
        ok = True
        for j in range(length):
            x = src[i - j]
            if is_na(x):
                ok = False
                break
            s += x * (length - j)
        if ok:
            out[i] = s / denom
    return out


def _stdev(src: list[float], length: int) -> list[float]:
    # TV ta.stdev: population stdev over the window
    out = [NAN] * len(src)
    for i in range(length - 1, len(src)):
        w = src[i - length + 1:i + 1]
        if any(is_na(x) for x in w):
            continue
        m = sum(w) / length
        out[i] = math.sqrt(sum((x - m) ** 2 for x in w) / length)
    return out


def _change(src: list[float], length: int = 1) -> list[float]:
    out = [NAN] * len(src)
    for i in range(length, len(src)):
        if not is_na(src[i]) and not is_na(src[i - length]):
            out[i] = src[i] - src[i - length]
    return out


def _ffill_by_date(chart_dates, value_map):
    """Align a date->value map onto chart bars, forward-filling gaps."""
    out = [NAN] * len(chart_dates)
    last = NAN
    for i, d in enumerate(chart_dates):
        v = value_map.get(d)
        if v is not None:
            last = v
        out[i] = last
    return out

def compute_signals(chart: data.Bars, symbol: str, p: dict) -> dict:
    n = len(chart)
    o, h, l, c, v = chart.open, chart.high, chart.low, chart.close, chart.volume
    dates = [_date(t) for t in chart.ts]

    # --- aux feeds (daily, date-aligned, ffilled) ---
    nya = data.get_daily("^NYA")
    nya_close = {_date(nya.ts[i]): nya.close[i] for i in range(len(nya))}
    nya_lag50 = {_date(nya.ts[i]): nya.close[i - 50]
                 for i in range(50, len(nya))}
    vixb = data.get_daily("^VIX")
    vix_map = {_date(vixb.ts[i]): vixb.close[i] for i in range(len(vixb))}
    nya_c = _ffill_by_date(dates, nya_close)
    nya_c50 = _ffill_by_date(dates, nya_lag50)
    vix = _ffill_by_date(dates, vix_map)

    breadth = _load_breadth()      # date -> (mcclellan, nh_pct, nl_pct, ratio_ok)
    br_al: list = [None] * n
    last_b = None
    for i, d in enumerate(dates):
        tup = breadth.get(d)
        if tup is not None:
            last_b = tup
        br_al[i] = last_b

    is_hind = [False] * n
    for i in range(n):
        tup = br_al[i]
        if tup is None or is_na(nya_c[i]) or is_na(nya_c50[i]):
            continue
        mcc, nh_pct, nl_pct, ratio_ok = tup
        is_hind[i] = (mcc < 0 and nh_pct >= p["thresh_pct"]
                      and nl_pct >= p["thresh_pct"] and ratio_ok
                      and nya_c[i] > nya_c50[i])

    # --- physics engine: velocity / acceleration / volume z (daily) ---
    vel = _wma(ta.roc(c, p["len_vel"]), 3)
    acc = ta.sma(_change(vel, p["len_acc"]), p["smooth_acc"])
    vmean18 = ta.sma(v, p["len_vel"])
    vstd18 = _stdev(v, p["len_vel"])
    high_vol = [False] * n
    for i in range(n):
        if not is_na(vmean18[i]) and not is_na(vstd18[i]) and vstd18[i] != 0:
            high_vol[i] = (v[i] - vmean18[i]) / vstd18[i] > 2.0

    # --- regimes (stateful crash confirmation) ---
    reg_crash = [False] * n
    reg_liquid = [False] * n
    reg_bear = [False] * n
    reg_power = [False] * n
    reg_decel = [False] * n
    counter = 0
    for i in range(n):
        raw = p["use_hinden"] and is_hind[i] and (not is_na(vel[i]) and vel[i] < 0)
        counter = counter + 1 if raw else 0
        crash = counter >= p["regime_confirm_bars"]
        decel = (not is_na(vel[i]) and vel[i] > 0
                 and not is_na(acc[i]) and acc[i] < 0)
        liquid = (not crash and not is_na(vel[i]) and vel[i] < 0
                  and not is_na(acc[i]) and acc[i] < 0 and high_vol[i])
        bear = (not crash and not liquid and not decel
                and not is_na(vel[i]) and vel[i] < 0)
        power = (not is_na(vel[i]) and vel[i] > 0
                 and not is_na(acc[i]) and acc[i] > 0 and not decel)
        reg_crash[i], reg_liquid[i], reg_bear[i] = crash, liquid, bear
        reg_power[i], reg_decel[i] = power, decel

    # --- institutional S/D zones (events only; boxes are display) ---
    avg_vol20 = ta.sma(v, 20)
    pl_ = p["piv_len"]
    ph_px, pl_px = ta.pivots(h, pl_, pl_)
    _, pl_lo = ta.pivots(l, pl_, pl_)
    new_dem = [False] * n
    new_sup = [False] * n
    for i in range(n):
        j = i - pl_
        if j < 0 or is_na(avg_vol20[j]):
            continue
        vol_ok = v[j] > avg_vol20[j] * p["vol_thresh"]
        if not is_na(pl_lo[i]) and vol_ok:
            new_dem[i] = True
        if not is_na(ph_px[i]) and vol_ok:
            new_sup[i] = True

    # --- Lumina TSI bear divergence (daily) ---
    pc = _change(c, 1)
    abs_pc = [abs(x) if not is_na(x) else NAN for x in pc]
    ds_pc = ta.ema(ta.ema(pc, p["lum_long_len"]), p["lum_short_len"])
    ds_abs = ta.ema(ta.ema(abs_pc, p["lum_long_len"]), p["lum_short_len"])
    tsi = [100.0 * ds_pc[i] / ds_abs[i]
           if not is_na(ds_pc[i]) and not is_na(ds_abs[i]) and ds_abs[i] != 0
           else NAN for i in range(n)]
    lb = p["lum_lookback"]
    tsi_ph, _ = ta.pivots(tsi, lb, lb)
    lumina = [False] * n
    prev_piv = None                    # (tsi_at_pivot, price_high_at_pivot)
    for i in range(n):
        if is_na(tsi_ph[i]):
            continue
        cur = (tsi[i - lb], h[i - lb])
        if prev_piv is not None:
            lumina[i] = cur[1] > prev_piv[1] and cur[0] < prev_piv[0]
        prev_piv = cur

    # --- RSI wick divergences (fractal, 2-bar confirmation) ---
    src_bull = l if p["rsi_wick"] else c
    src_bear = h if p["rsi_wick"] else c
    rsi_bull = ta.rsi(src_bull, p["len_rsi"])
    rsi_bear = ta.rsi(src_bear, p["len_rsi"])
    rsi_classic = ta.rsi(c, 14)

    def fractal(src, i, top):
        if i < 4 or any(is_na(src[i - j]) for j in range(5)):
            return False
        a, b_, mid, d_, e = src[i - 4], src[i - 3], src[i - 2], src[i - 1], src[i]
        if top:
            return a < mid and b_ < mid and mid > d_ and mid > e
        return a > mid and b_ > mid and mid < d_ and mid < e

    rsi_hid_bull = [False] * n
    rsi_reg_bull = [False] * n
    rsi_reg_bear = [False] * n
    prev_top = None                    # (rsi, price) at previous fractal top
    prev_bot = None
    for i in range(n):

            cur = (rsi_bear[i - 2], src_bear[i - 2])
            if prev_top is not None:
                rsi_reg_bear[i] = cur[1] > prev_top[1] and cur[0] < prev_top[0]
            prev_top = cur
        if fractal(rsi_bull, i, False):
            cur = (rsi_bull[i - 2], src_bull[i - 2])
            if prev_bot is not None:
                rsi_hid_bull[i] = cur[1] > prev_bot[1] and cur[0] < prev_bot[0]
                rsi_reg_bull[i] = cur[1] < prev_bot[1] and cur[0] > prev_bot[0]
            prev_bot = cur

    # --- PPO bull divergence ---
    fast = ta.ema(c, p["ppo_fast"])
    slow = ta.ema(c, p["ppo_slow"])
    macd_pct = [(fast[i] - slow[i]) / slow[i] * 100.0
                if not is_na(fast[i]) and not is_na(slow[i]) and slow[i] != 0
                else NAN for i in range(n)]
    d_sig = ta.sma(macd_pct, p["ppo_smooth"])
    ppo_bull = [False] * n
    bars_since_pmin = 10 ** 9
    cur_plow = prev_plow = NAN
    cur_olow = prev_olow = NAN
    for i in range(2, n):
        price_min = l[i] > l[i - 1] and l[i - 1] < l[i - 2]
        osc_min = (not is_na(d_sig[i]) and not is_na(d_sig[i - 1])
                   and not is_na(d_sig[i - 2])
                   and d_sig[i] > d_sig[i - 1] and d_sig[i - 1] < d_sig[i - 2])
        if price_min:
            prev_plow, cur_plow = cur_plow, l[i - 1]
            bars_since_pmin = 0
        else:
            bars_since_pmin += 1
        if osc_min:
            prev_olow, cur_olow = cur_olow, d_sig[i - 1]
        if (osc_min and bars_since_pmin <= 5
                and not is_na(prev_plow) and not is_na(prev_olow)):
            ppo_bull[i] = cur_plow < prev_plow and cur_olow > prev_olow

    # --- gap logic (regime gate applied in the sim loop) ---
    is_gap = [False] * n
    gap_filled = [False] * n
    for i in range(1, n):
        gap_size = (o[i] - h[i - 1]) / h[i - 1] * 100.0
        if o[i] > h[i - 1] and gap_size >= p["gap_thresh"]:
            is_gap[i] = True
            gap_filled[i] = l[i] <= h[i - 1]

    # --- volume blow-off confluence ---
    vmean20 = ta.sma(v, 20)
    vstd20 = _stdev(v, 20)
    vmax20 = ta.rolling_max(v, 20)
    vol_score = [0] * n
    for i in range(n):
        z = ((v[i] - vmean20[i]) / vstd20[i]
             if not is_na(vmean20[i]) and not is_na(vstd20[i]) and vstd20[i] != 0
             else 0.0)
        if i >= 50:
            rank = sum(1 for j in range(1, 51) if v[i] > v[i - j])
            pctl = rank / 50.0 * 100.0
        else:
            pctl = 0.0
        peak = v[i] / vmax20[i] if not is_na(vmax20[i]) and vmax20[i] != 0 else 0.0
        s = (3 if z > 2.5 else 2 if z > 2.0 else 1 if z > 1.5 else 0)
        s += (3 if pctl > 95 else 2 if pctl > 90 else 1 if pctl > 85 else 0)
        s += (3 if peak > 0.90 else 2 if peak > 0.80 else 1 if peak > 0.70 else 0)
        vol_score[i] = s
    mom5 = [((c[i] - c[i - 5]) / c[i - 5] * 100.0) if i >= 5 else 0.0
            for i in range(n)]
